### 专题一 · 顺序表（5 个题型）

**题型 1 查找** —— 一趟扫描 + 一个记录变量

```c
int findX(int A[], int n) {
    int best = A[0];
    for (int i = 1; i < n; i++)
        if (A[i] > best) best = A[i];   // 条件按题意换
    return best;
}
```

**题型 2 排序 / 题型 3 合并** —— 双指针归并

```c
void merge(int A[], int m, int B[], int n, int C[]) {
    int i = 0, j = 0, k = 0;
    while (i < m && j < n)
        C[k++] = (A[i] <= B[j]) ? A[i++] : B[j++];   // <= 保证稳定
    while (i < m) C[k++] = A[i++];
    while (j < n) C[k++] = B[j++];
}
```

**题型 4 双指针** —— 三种变体，认准用哪一种

| 变体     | 指针怎么动                               | 典型题                    |
| -------- | ---------------------------------------- | ------------------------- |
| 快慢指针 | 一个走一步，一个走两步                   | 找中点、判环、倒数第 k 个 |
| 前后指针 | 一头一尾向中间夹                         | 逆置、两数之和            |
| 读写指针 | 读指针走遍全表，写指针只在满足条件时前进 | 删除元素、去重            |

```c
// 读写双指针：删除所有等于 x 的元素，O(n) 一趟，O(1) 空间
int del_x(int A[], int n, int x) {
    int k = 0;                    // k 既是写指针，也是新长度
    for (int i = 0; i < n; i++)   // i 是读指针
        if (A[i] != x) A[k++] = A[i];
    return k;
}
```

**题型 5 删除元素** —— 就是上面的读写双指针。**不要写"找到一个就把后面全体前移"**，那是 O(n²)。

**通用逆置（3 行，2010 真题核心）**

```c
void Reverse(int A[], int from, int to) {
    for (int i = 0; i < (to - from + 1) / 2; i++) {
        int t = A[from + i]; A[from + i] = A[to - i]; A[to - i] = t;
    }
}
```

### 专题二 · 链表（5 个题型）

**通用前提**：带头结点。所有遍历从 `p = L->next` 开始，删除要靠**前驱指针**。

**题型 1 排序** —— 链表上的插入排序

```c
void sortList(LinkList L) {
    LNode *p = L->next, *pre, *r;
    if (!p) return;
    r = p->next; p->next = NULL; p = r;       // 断开，只留第一个作为有序区
    while (p) {
        r = p->next;                           // 先存后继，否则丢链
        pre = L;
        while (pre->next && pre->next->data < p->data) pre = pre->next;
        p->next = pre->next; pre->next = p;    // 插入
        p = r;
    }
}
```

**题型 2 逆置（头插法，最重要的一个手法）**

```c
void reverseList(LinkList L) {
    LNode *p = L->next, *r;
    L->next = NULL;                 // 摘掉整条链
    while (p) {
        r = p->next;                // 必须先存后继
        p->next = L->next;          // 头插
        L->next = p;
        p = r;
    }
}
```

> 三行顺序是死的：**存后继 → 改指针 → 走下一个**。不存 r 就断链。

**题型 3 合并**

```c
void mergeList(LinkList La, LinkList Lb) {   // 归并到 La，两表均升序
    LNode *pa = La->next, *pb = Lb->next, *rc = La;
    while (pa && pb) {
        if (pa->data <= pb->data) { rc->next = pa; rc = pa; pa = pa->next; }
        else                      { rc->next = pb; rc = pb; pb = pb->next; }
    }
    rc->next = pa ? pa : pb;
    free(Lb);
}
```

**题型 4 快慢指针**

```c
LNode *findMid(LinkList L) {                 // 中间结点
    LNode *slow = L->next, *fast = L->next;
    while (fast && fast->next) { slow = slow->next; fast = fast->next->next; }
    return slow;
}
LNode *findKthFromEnd(LinkList L, int k) {   // 倒数第 k 个（2009 真题）
    LNode *p = L->next, *q = L->next; int cnt = 0;
    while (p) { if (cnt < k) cnt++; else q = q->next; p = p->next; }
    return (cnt < k) ? NULL : q;
}
```

**题型 5 删除**

```c
void delNode(LinkList L, ElemType x) {
    LNode *pre = L, *p = L->next, *r;
    while (p) {
        if (p->data == x) { r = p; pre->next = p->next; p = p->next; free(r); }
        else              { pre = p; p = p->next; }
    }
}
```

> **删除后 `pre` 不能前进**。这是链表题最高频的错。

### 专题三 · 栈和队列

**手法 1 用栈把递归改成非递归**

```c
void inOrderNoRec(BiTree T) {          // 中序非递归模板
    BiTNode *st[MaxSize]; int top = -1;
    BiTNode *p = T;
    while (p || top != -1) {
        if (p) { st[++top] = p; p = p->lchild; }     // 一路向左压栈
        else   { p = st[top--]; visit(p); p = p->rchild; }
    }
}
```

**手法 2 括号匹配 / 表达式求值**：左括号入栈，右括号出栈比对，结束时栈必须为空。

**手法 3 循环队列**

```c
int enQueue(SqQueue *Q, ElemType x) {
    if ((Q->rear + 1) % MaxSize == Q->front) return 0;   // 满
    Q->data[Q->rear] = x; Q->rear = (Q->rear + 1) % MaxSize; return 1;
}
int deQueue(SqQueue *Q, ElemType *x) {
    if (Q->front == Q->rear) return 0;                    // 空
    *x = Q->data[Q->front]; Q->front = (Q->front + 1) % MaxSize; return 1;
}
```

### 专题四 · 树（5 个题型）

**题型 1 层次遍历解决的问题** —— 树高、每层最大值、完全二叉树判定、树的宽度

```c
int treeDepth(BiTree T) {
    if (!T) return 0;
    BiTNode *Q[MaxSize]; int front = 0, rear = 0, depth = 0;
    Q[rear++] = T;
    while (front < rear) {
        int size = rear - front;                // 本层结点数：分层的关键
        depth++;
        while (size--) {
            BiTNode *p = Q[front++];
            if (p->lchild) Q[rear++] = p->lchild;
            if (p->rchild) Q[rear++] = p->rchild;
        }
    }
    return depth;
}
```

> `int size = rear - front;` 这一行是整个层序题型的命门，没有它就分不出层。

**题型 2 递归** —— 三段式模板，90% 的树题都是它

```
f(T):
    ① if (!T) return 边界值;          // 递归出口
    ② 左 = f(T->lchild); 右 = f(T->rchild);   // 拿子问题的解
    ③ return 由 T->data、左、右 组合出答案;
```

```c
int countLeaf(BiTree T) {
    if (!T) return 0;
    if (!T->lchild && !T->rchild) return 1;
    return countLeaf(T->lchild) + countLeaf(T->rchild);
}
int WPL(BiTree T, int d) {                         // 带权路径长度（2014 真题）
    if (!T) return 0;
    if (!T->lchild && !T->rchild) return T->weight * d;
    return WPL(T->lchild, d + 1) + WPL(T->rchild, d + 1);
}
```

**题型 3 求祖先** —— 用**后序非递归**，因为后序栈里恰好装着当前结点的全部祖先

```c
void ancestors(BiTree T, BiTNode *p) {
    struct { BiTNode *t; int tag; } st[MaxSize];   // tag=0 左已访，tag=1 右已访
    int top = 0;
    BiTNode *bt = T;
    while (bt || top > 0) {
        while (bt) { st[++top].t = bt; st[top].tag = 0; bt = bt->lchild; }
        while (top > 0 && st[top].tag == 1) {
            if (st[top].t == p) { for (int i = 1; i < top; i++) visit(st[i].t); return; }
            top--;
        }
        if (top > 0) { st[top].tag = 1; bt = st[top].t->rchild; }
    }
}
```

### 专题五 · 图（3 个题型）

**通用骨架**

```c
int visited[MaxSize];
void DFS(MGraph G, int v) {
    visited[v] = 1; visit(v);
    for (int w = 0; w < G.numVertex; w++)
        if (G.Edge[v][w] && !visited[w]) DFS(G, w);
}
void BFS(MGraph G, int v) {
    int Q[MaxSize], f = 0, r = 0;
    visited[v] = 1; visit(v); Q[r++] = v;
    while (f < r) {
        int u = Q[f++];
        for (int w = 0; w < G.numVertex; w++)
            if (G.Edge[u][w] && !visited[w]) { visited[w] = 1; visit(w); Q[r++] = w; }
    }
}
```

**题型 1 求路径（DFS + 回溯）** —— 图题的核心手法

```c
int path[MaxSize], d = -1;
void findPath(MGraph G, int u, int v) {
    visited[u] = 1; path[++d] = u;
    if (u == v) { for (int i = 0; i <= d; i++) printf("%d ", path[i]); printf("\n"); }
    else
        for (int w = 0; w < G.numVertex; w++)
            if (G.Edge[u][w] && !visited[w]) findPath(G, w, v);
    visited[u] = 0; d--;             // 回溯：撤销标记 + 弹出路径
}
```

> **回溯就是最后那两行"撤销"**。忘了撤销只能找到一条路径；只撤 visited 不撤 d，路径会错乱。

**题型 2 连通与有环**

- 无向图连通分量数 = 从每个未访问顶点发起 DFS 的次数
- 有向图判环 = 拓扑排序后仍有顶点没输出
- 无向图判环 = DFS 时遇到已访问且非父结点的邻居

**题型 3 存储结构转换** —— 邻接矩阵 ↔ 邻接表，双重循环扫矩阵，非零就头插到链表

**求度（邻接矩阵）**

```c

int partition(int A[], int low, int high) {
    int pivot = A[low];
    while (low < high) {
        while (low < high && A[high] >= pivot) high--;
        A[low] = A[high];
        while (low < high && A[low]  <= pivot) low++;
        A[high] = A[low];
```

```c
int Search_k_bf(LinkList list, int k) {
    int len = 0;
    LNode *p = list->next;
    while (p) { len++; p = p->next; }          // 第一趟：数长度
```

```c
int Search_k(LinkList list, int k) {
```

### 2010 · 42（13 分）数组循环左移 p 位

> R[0..n-1] 存 n 个整数，把序列循环左移 p 位（0 < p < n），时间空间都尽可能高效。

**暴力解 O(n) 时间 / O(p) 空间**

```c
void Converse_bf(int R[], int n, int p) {
    int *S = (int *)malloc(sizeof(int) * p);
    for (int i = 0; i < p; i++) S[i] = R[i];            // 暂存前 p 个
    for (int i = p; i < n; i++) R[i - p] = R[i];        // 后 n-p 个左移
    for (int i = 0; i < p; i++) R[n - p + i] = S[i];    // 放回
    free(S);
}
```

**最优解 O(n) 时间 / O(1) 空间 —— 三次逆置**

> 思路：把 `ab` 变成 `ba`。先 a 逆置得 `a⁻¹b`，再 b 逆置得 `a⁻¹b⁻¹`，最后整体逆置得 `(a⁻¹b⁻¹)⁻¹ = ba`。

```c
void Reverse(int R[], int from, int to) {
    for (int i = 0; i < (to - from + 1) / 2; i++) {
        int t = R[from + i]; R[from + i] = R[to - i]; R[to - i] = t;
    }
}
void Converse(int R[], int n, int p) {
    Reverse(R, 0, p - 1);
    Reverse(R, p, n - 1);
    Reverse(R, 0, n - 1);
}
```

**踩坑**：三次逆置的顺序不能乱；`(to-from+1)/2` 保证奇数长度时中间元素不动。

### 2011 · 42（15 分）两个等长升序序列的中位数

> A、B 各含 n 个升序整数，求两序列合起来后的中位数（第 ⌈2n/2⌉ = 第 n 小）。时间空间都尽可能高效。

**暴力解 O(n) 时间 / O(1) 空间 —— 归并计数**

```c
int M_Search_bf(int A[], int B[], int n) {
    int i = 0, j = 0, cnt = 0, cur = 0;
    while (cnt < n) {                      // 只需要走到第 n 个
        cur = (A[i] <= B[j]) ? A[i++] : B[j++];
        cnt++;
    }
    return cur;
}
```

> 这个解已经能拿到相当分数（O(n)）。但题目说"时间和空间**两方面**都尽可能高效"，暗示还有更好的。

**最优解 O(log n) 时间 / O(1) 空间 —— 双序列二分**

```c
int M_Search(int A[], int B[], int n) {
    int s1 = 0, d1 = n - 1, m1;      // A 的查找区间
    int s2 = 0, d2 = n - 1, m2;      // B 的查找区间
    while (s1 != d1 || s2 != d2) {
        m1 = (s1 + d1) / 2;
        m2 = (s2 + d2) / 2;
        if (A[m1] == B[m2]) return A[m1];        // 相等即为中位数
        if (A[m1] < B[m2]) {                     // 舍弃 A 的前半、B 的后半
            if ((s1 + d1) % 2 == 0) { s1 = m1;     d2 = m2; }   // 元素个数为奇数
            else                    { s1 = m1 + 1; d2 = m2; }   // 元素个数为偶数
        } else {                                 // 舍弃 A 的后半、B 的前半
            if ((s1 + d1) % 2 == 0) { d1 = m1; s2 = m2;     }
            else                    { d1 = m1; s2 = m2 + 1; }
        }
    }
    return A[s1] < B[s2] ? A[s1] : B[s2];
}
```

**踩坑**：奇偶分支是这题唯一的难点。记忆口诀——**"舍弃时，含中位数的那一侧要保留中间元素"**：区间元素个数为奇数时保留 mid，为偶数时可以舍掉 mid。写不出来就交暴力解 O(n)。

### 2012 · 42（15 分）两个单链表的公共后缀起始位置

> "loading" 和 "being" 共享后缀 "ing"。str1、str2 为两条带头结点单链表的头指针，找出公共后缀的第一个结点。

**暴力解 O(mn)**

```c
LNode *findAddr_bf(LinkList str1, LinkList str2) {
    for (LNode *p = str1->next; p; p = p->next)
        for (LNode *q = str2->next; q; q = q->next)
            if (p == q) return p;         // 比的是指针（地址），不是 data

}
```

```c
int listLen(LinkList L) {
    int len = 0;
    for (LNode *p = L->next; p; p = p->next) len++;
    return len;
}
LNode *findAddr(LinkList str1, LinkList str2) {
    int m = listLen(str1), n = listLen(str2);
```

### 2013 · 41（15 分）找主元素

> A[0..n-1] 中元素取值 0~n-1。若某元素出现次数 > n/2 则为主元素，返回它；否则返回 -1。

**暴力解 O(n²)** —— 评分表：O(n²) 给 8 分左右

```c
int Majority_bf(int A[], int n) {
    for (int i = 0; i < n; i++) {
        int cnt = 0;
        for (int j = 0; j < n; j++) if (A[j] == A[i]) cnt++;
        if (cnt > n / 2) return A[i];
    }
    return -1;
}
```

**次优解 O(n) 时间 / O(n) 空间 —— 计数排序思想**（参考答案明写此解给 10/11 分）

```c
int Majority_cnt(int A[], int n) {
    int *p = (int *)malloc(sizeof(int) * n);
    for (int k = 0; k < n; k++) p[k] = 0;
    int max = 0;
    for (int k = 0; k < n; k++) {
        p[A[k]]++;
        if (p[A[k]] > p[max]) max = A[k];
    }
    int res = (p[max] > n / 2) ? max : -1;
    free(p);
    return res;
}
```

**最优解 O(n) 时间 / O(1) 空间 —— 摩尔投票（满分 15）**

```c
int Majority(int A[], int n) {
    int c = A[0], count = 1;             // c 是候选主元素
    for (int i = 1; i < n; i++) {        // 第一趟：选候选
        if (A[i] == c) count++;
        else {
            count--;
            if (count == 0) { c = A[i]; count = 1; }   // 换候选，重新计数
        }
    }
    count = 0;
    for (int i = 0; i < n; i++) if (A[i] == c) count++;  // 第二趟：验证
    return (count > n / 2) ? c : -1;
}
```

**踩坑**：**第二趟验证不能省**。摩尔投票只保证"若存在主元素则一定是 c"，不保证 c 一定是主元素。省了这一趟，输入 `{1,2,3}` 就会错误返回 3。

### 2014 · 41（13 分）二叉树的带权路径长度 WPL

> 结点结构 `left | weight | right`，叶结点的 weight 是非负权值。求 WPL = Σ(叶结点权值 × 其深度)。
> **本题明确要求"给出二叉树结点的数据类型定义"——必须写。**

```c
typedef struct BiTNode {
    int weight;
    struct BiTNode *left, *right;
} BiTNode, *BiTree;
```

**解法一：递归（推荐，最短）**

```c
int wpl_PreOrder(BiTree root, int deep) {
    if (root == NULL) return 0;
    if (root->left == NULL && root->right == NULL)   // 叶结点
        return root->weight * deep;
    return wpl_PreOrder(root->left,  deep + 1)
         + wpl_PreOrder(root->right, deep + 1);
}
int WPL(BiTree root) { return wpl_PreOrder(root, 0); }
```

**解法二：层序遍历（不想递归时用）**

```c
int WPL_level(BiTree root) {
    if (!root) return 0;
    BiTNode *Q[MaxSize]; int front = 0, rear = 0;
    int deep = 0, wpl = 0;
    Q[rear++] = root;
    while (front < rear) {
        int size = rear - front;
        while (size--) {
            BiTNode *p = Q[front++];
            if (!p->left && !p->right) wpl += p->weight * deep;
            if (p->left)  Q[rear++] = p->left;
            if (p->right) Q[rear++] = p->right;
        }
        deep++;
    }
    return wpl;
}
```

**踩坑**：根结点深度是 **0**（WPL 定义里根到根的路径长度为 0）。写成 1 会整体偏大。

### 2015 · 41（15 分）删除单链表中绝对值相等的重复结点

> 带头结点单链表，|data| ≤ n。对每个绝对值只保留**第一次**出现的结点，删除其余。要求时间尽可能高效。

**暴力解 O(n²) / O(1) 空间**

```c
void func_bf(PNODE h) {
    NODE *p = h->link;
    while (p) {
        NODE *pre = p, *q = p->link;
        while (q) {
            int a = p->data > 0 ? p->data : -p->data;
            int b = q->data > 0 ? q->data : -q->data;
            if (a == b) { pre->link = q->link; free(q); q = pre->link; }
            else        { pre = q; q = q->link; }
        }
        p = p->link;
    }
}
```

**最优解 O(m) 时间 / O(n) 空间 —— 空间换时间**

```c
void func(PNODE h, int n) {
    PNODE p = h, r;
    int *q = (int *)malloc(sizeof(int) * (n + 1));   // 辅助数组，下标 0..n
    for (int i = 0; i < n + 1; i++) *(q + i) = 0;    // 全置 0
    while (p->link != NULL) {
        int m = p->link->data > 0 ? p->link->data : -p->link->data;   // 取绝对值
        if (*(q + m) == 0) {          // 首次出现 → 保留
            *(q + m) = 1;
            p = p->link;
        } else {                       // 重复出现 → 删除
            r = p->link;
            p->link = r->link;
            free(r);
        }
    }
    free(q);
}
```

**踩坑**：辅助数组大小是 **n+1** 不是 n，因为 |data| 可以取到 n，要用下标 n。参考答案原文用的就是 `*(q+i)` 这种指针写法，等价于 `q[i]`。

### 2016 · 43（13 分）把集合划分成两个子集

> n 个整数分成 A1(n1 个) 和 A2(n2 个)，使 |n1 − n2| 最小、|S1 − S2| 最大（S 为各自元素和）。返回 S2 − S1。

**暴力解 O(n log n)** —— 全排序后前一半给 A1（评分表给 11 分）

```c
int setPartition_bf(int a[], int n) {
    // 任意 O(n log n) 排序，此处用简单写法示意
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (a[j] > a[j+1]) { int t = a[j]; a[j] = a[j+1]; a[j+1] = t; }   // 这样是 O(n^2)
    int k = n / 2, s1 = 0, s2 = 0;
    for (int i = 0; i < k; i++) s1 += a[i];
    for (int i = k; i < n; i++) s2 += a[i];
    return s2 - s1;
}
```

**最优解 O(n) 平均 / O(1) 空间 —— 快排划分，不全排序（满分 13）**

> 关键洞察：**只需要"最小的一半"和"最大的一半"分开，不需要各自内部有序**。所以只要用快排的 partition 反复缩小，直到枢轴恰好落在第 n/2 个位置。

```c
int setPartition(int a[], int n) {
    int pivot, low = 0, low0 = 0, high = n - 1, high0 = n - 1;
    int flag = 1, k = n / 2, i, s1 = 0, s2 = 0;
    while (flag) {
        pivot = a[low];
        while (low < high) {                                   // 一次划分
            while (low < high && a[high] >= pivot) --high;
            if (low != high) a[low] = a[high];
            while (low < high && a[low] <= pivot) ++low;
            if (low != high) a[high] = a[low];
        }
        a[low] = pivot;
        if (low == k - 1) flag = 0;                            // 枢轴恰在分界点，结束
        else if (low < k - 1) { low0 = ++low; high = high0; }  // 只在右半边继续划分
        else                  { high0 = --high; low = low0; }  // 只在左半边继续划分
    }
    for (i = 0; i < k; i++) s1 += a[i];
    for (i = k; i < n; i++) s2 += a[i];
    return s2 - s1;
}
```

**加分解 O(n) 时间 / O(1) 空间 —— 原地置换**（超出参考答案，写对是加分项）

```c
int findMissMin_O1(int A[], int n) {

        while (A[i] > 0 && A[i] <= n && A[A[i] - 1] != A[i]) {
```

### 2019 · 41（13 分）单链表重排

> L = (a₁,a₂,…,aₙ) → L′ = (a₁,aₙ,a₂,aₙ₋₁,a₃,aₙ₋₂,…)。**空间复杂度必须是 O(1)**，时间尽可能高效。

**暴力解 O(n) 时间 / O(n) 空间**（不满足题目要求，但比空着强）

```c
void change_bf(LinkList L) {
    LNode *arr[MaxSize]; int n = 0;
    for (LNode *p = L->next; p; p = p->next) arr[n++] = p;
    int i = 0, j = n - 1;
    LNode *tail = L;
    while (i <= j) {
        tail->next = arr[i]; tail = arr[i]; i++;
        if (i <= j) { tail->next = arr[j]; tail = arr[j]; j--; }
    }
    tail->next = NULL;
}
```

> 题目白纸黑字要求 O(1) 空间，这个解会被扣分，但**算法思想部分仍能得分**。

**最优解 O(n) 时间 / O(1) 空间 —— 找中点 + 后半逆置 + 交叉合并**

```c
void change(LinkList L) {
    LNode *p = L, *q = L, *r, *s;
    /* ① 快慢指针找中间结点，p 停在前半段最后一个 */
    while (q->next != NULL) {
        p = p->next;
        q = q->next;
        if (q->next != NULL) q = q->next;   // 这一行不能写成 q = q->next->next
    }
    /* ② 断开，把后半段用头插法逆置到 p 之后 */
    q = p->next;
    p->next = NULL;
    while (q != NULL) {
        r = q->next;
        q->next = p->next;
        p->next = q;
        q = r;
    }
    /* ③ 前半段 s 与逆置后的后半段 q 交叉插入 */
    s = L->next;
    q = p->next;
    p->next = NULL;
    while (q != NULL) {
        r = q->next;
        q->next = s->next;
        s->next = q;
        s = q->next;
        q = r;
    }
}
```

**踩坑**（这一条是本手册唯一一处编译验证时**真的崩溃**过的地方，务必看）：

1. **找中点的写法必须让前半段拿到多出来的那个结点。** 写成 `while (q->next && q->next->next) q = q->next->next;` 时，n=7 会让前半段只有 3 个、后半段有 4 个，第三步交叉插入到最后 `s` 会先变成 NULL，然后 `s->next` 空指针崩溃。正确写法是上面那种「先走一步，能走再走一步」，保证前半段 ≥ 后半段。
2. 三步顺序不能乱，且每一步都要先用 `r` 存后继。
3. 这题是**链表所有手法的集大成**：快慢指针 + 头插逆置 + 交叉合并，三个手法一道题里全考了。练熟这一题等于练熟整个链表专题。

### 2020 · 41（13 分）三元组的最小距离

> 三个非空升序整数集 S1、S2、S3 分别存在数组 A(n)、B(m)、C(p) 中。定义 D = |a−b| + |b−c| + |c−a|，求最小的 D。

**暴力解 O(n·m·p)**

```c
int findMinofTrip_bf(int A[], int n, int B[], int m, int C[], int p) {
    int D_min = 0x7fffffff;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            for (int k = 0; k < p; k++) {
                int D = abs(A[i]-B[j]) + abs(B[j]-C[k]) + abs(C[k]-A[i]);
                if (D < D_min) D_min = D;
            }
    return D_min;
}
```

**最优解 O(n+m+p) 时间 / O(1) 空间 —— 三指针，每次只移动最小值那个**

> 数学基础：设 a ≤ b ≤ c，则 D = (b−a) + (c−b) + (c−a) = **2(c−a)**。所以 D 只由最大最小值的跨度决定。要缩小跨度，只能把**最小的那个往右移**（右移最大的只会让跨度更大）。

```c
#define INT_MAX 0x7fffffff
int abs_(int a) { return a < 0 ? -a : a; }
int is_min(int a, int b, int c) { return a <= b && a <= c; }   // a 是否为三者最小

int findMinofTrip(int A[], int n, int B[], int m, int C[], int p) {
    int i = 0, j = 0, k = 0, D_min = INT_MAX, D;
    while (i < n && j < m && k < p && D_min > 0) {
        D = abs_(A[i]-B[j]) + abs_(B[j]-C[k]) + abs_(C[k]-A[i]);   // 计算 D
        if (D < D_min) D_min = D;                                   // 更新
        if      (is_min(A[i], B[j], C[k])) i++;                     // 谁最小谁前进
        else if (is_min(B[j], C[k], A[i])) j++;
        else                               k++;
    }
    return D_min;
}
```

**踩坑**：`D_min > 0` 这个提前退出条件是加分点（D=0 已是理论最小，没必要继续）。三指针题的通用问法都是"移动哪一个"，答案永远是**移动最小的那个**。

### 2021 · 41（15 分）判断无向图是否存在 EL 路径

> 无向连通图 G 用邻接矩阵存储。当 G 中**度为奇数的顶点个数为不大于 2 的偶数**时，存在包含所有边的 EL 路径。设计 `int IsExistEL(MGraph G)`，存在返回 1 否则返回 0。

> 非空二叉树 T 结点值均为正整数，顺序存储：`int SqBiTNode[MAX_SIZE]; int ElemNum;`。设计尽可能高效的算法，是 BST 返回 true，否则返回 false。

**暴力解 O(n²)** —— 对每个结点检查它整棵左子树 < 它 < 整棵右子树

```c
int getMax(SqBiTree bt, int i) { /* 递归求以 i 为根的子树最大值，略 */ }
int getMin(SqBiTree bt, int i) { /* 同理 */ }
int judgeBST_bf(SqBiTree bt, int i) {
    if (i >= bt.ElemNum || bt.SqBiTNode[i] == 0) return 1;
    int L = 2*i + 1, R = 2*i + 2;
    if (L < bt.ElemNum && bt.SqBiTNode[L] && getMax(bt, L) >= bt.SqBiTNode[i]) return 0;
    if (R < bt.ElemNum && bt.SqBiTNode[R] && getMin(bt, R) <= bt.SqBiTNode[i]) return 0;
    return judgeBST_bf(bt, L) && judgeBST_bf(bt, R);
}
```

**最优解 O(n) —— 中序遍历必须严格递增**

> BST 的定义等价于：**中序遍历序列严格递增**。顺序存储下，结点 i 的左右孩子是 2i+1、2i+2（下标从 0 开始），值为 0 表示该位置空。

```c
int pre = 0;                    // 记录中序遍历的前一个值（结点值均为正，初值 0 安全）

bool judgeBST(SqBiTree bt, int i) {
    if (i >= bt.ElemNum || bt.SqBiTNode[i] == 0) return true;   // 空结点，直接通过
    if (!judgeBST(bt, 2 * i + 1)) return false;                 // ① 递归判左子树
    if (bt.SqBiTNode[i] <= pre)   return false;                 // ② 当前值必须大于前驱
    pre = bt.SqBiTNode[i];                                      //    更新前驱
    return judgeBST(bt, 2 * i + 2);                             // ③ 递归判右子树
}
bool isBST(SqBiTree bt) { pre = 0; return judgeBST(bt, 0); }
```

**踩坑**：

1. **必须是全局/静态的 `pre`**，写成参数传值会失效。
2. 判等号：BST 不允许相等，所以是 `<=` 就返回 false。
3. 顺序存储下标从 0 开始时孩子是 `2i+1 / 2i+2`；从 1 开始时是 `2i / 2i+1`。**看题目给的定义决定**。

### 2024 · 41（13 分）判定有向图是否存在唯一的拓扑序列

> 有向图 G 邻接矩阵存储。设计 `int uniquely(MGraph G)`，若 G 存在唯一拓扑序列返回 1，否则返回 0。（背景：航天工程的子工程先导关系）

> **核心结论**：拓扑序列唯一 ⟺ **每一轮入度为 0 的顶点都恰好只有 1 个**。有 0 个说明有环，有 ≥2 个说明这一步可以任选，序列就不唯一。

```c
int uniquely(MGraph G) {
    int n = G.numVertices;
    int indeg[MAXV], removed[MAXV];

    for (int i = 0; i < n; i++) { indeg[i] = 0; removed[i] = 0; }
    for (int i = 0; i < n; i++)                       // 统计各顶点入度：第 j 列非零个数
        for (int j = 0; j < n; j++)
            if (G.Edge[i][j] != 0) indeg[j]++;

    for (int step = 0; step < n; step++) {            // 每轮删掉一个顶点
        int cnt = 0, v = -1;
        for (int i = 0; i < n; i++)
            if (!removed[i] && indeg[i] == 0) { cnt++; v = i; }
        if (cnt != 1) return 0;                       // 0 个 → 有环；多个 → 不唯一
        removed[v] = 1;                               // 删除顶点 v
        for (int j = 0; j < n; j++)                   // 及其所有出边
            if (G.Edge[v][j] != 0) indeg[j]--;
    }
    return 1;
}
```

- 时间复杂度 O(n²)，空间复杂度 O(n)

**踩坑**：不要真去做完整的拓扑排序再比较结果——**边排边判**，一发现 `cnt != 1` 立刻返回 0。另一种等价写法：拓扑序列中相邻两个顶点之间必须都有边。

### 2025 · 41（13 分）后缀最大乘积

> 两个长度为 n 的一维整型数组 A[n]、res[n]。计算 A[i] 与 A[j]（**0 ≤ i ≤ j ≤ n−1**）乘积的最大值，保存到 res[i]。
> 例：A = {1, 4, −9, 6} → res = {6, 24, 81, 36}。
> 函数原型：`void CalMulMax(int A[], int res[], int n)`

**暴力解 O(n²) 时间 / O(1) 空间（参考答案的"方法 1"）**

```c
void CalMulMax_bf(int A[], int res[], int n) {
    for (int i = 0; i < n; i++) {
        res[i] = A[i] * A[i];                       // j 可以等于 i，先取自乘
        for (int j = i + 1; j < n; j++)
            if (A[i] * A[j] > res[i]) res[i] = A[i] * A[j];
    }
}
```

**最优解 O(n) 时间 / O(1) 额外空间 —— 从右往左维护后缀最大值和最小值**

> 关键：A[i] 可能是负数。负数乘以**最小的负数**才最大。所以只维护后缀最大值不够，还要维护后缀最小值，两者取大。

```c
void CalMulMax(int A[], int res[], int n) {
    int mx = A[n-1], mn = A[n-1];        // A[i..n-1] 的最大值 / 最小值
    res[n-1] = A[n-1] * A[n-1];
    for (int i = n - 2; i >= 0; i--) {
        if (A[i] > mx) mx = A[i];        // 先把 A[i] 自己并入后缀区间（因为 j 可取 i）
        if (A[i] < mn) mn = A[i];
        int p1 = A[i] * mx;              // A[i] 为正时，乘后缀最大值最优
        int p2 = A[i] * mn;              // A[i] 为负时，乘后缀最小值最优
        res[i] = p1 > p2 ? p1 : p2;
    }
}
```

**验证**（用题目给的例子 A = {1, 4, −9, 6}）：

| i    | A[i] | mx   | mn   | A[i]*mx | A[i]*mn | res[i] | 期望 |
| ---- | ---- | ---- | ---- | ------- | ------- | ------ | ---- |
| 3    | 6    | 6    | 6    | 36      | 36      | 36     | 36 ✓ |
| 2    | −9   | 6    | −9   | −54     | 81      | 81     | 81 ✓ |
| 1    | 4    | 6    | −9   | 24      | −36     | 24     | 24 ✓ |
| 0    | 1    | 6    | −9   | 6       | −9      | 6      | 6 ✓  |

**踩坑**：

1. **j 可以等于 i**，所以 `res[i]` 的候选里包含 `A[i]*A[i]`。写成 `j = i+1` 起步且没先并入 A[i]，n=1 或全负数时会错。
2. **必须维护最小值**。只维护 mx 的话，A={−9, −9} 会算出 81 还是 −54？只有 mn 才能给出正确的 81。负数是这题的全部陷阱。
