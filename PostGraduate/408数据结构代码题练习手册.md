### 10.1 线性表（★★★ 全部默写）

```c
/* 顺序表：插入（在第 i 个位置，i 从 1 计）*/
int ListInsert(SqList *L, int i, ElemType e) {
    if (i < 1 || i > L->length + 1) return 0;      // 位置非法
    if (L->length >= MaxSize) return 0;            // 表满
    for (int j = L->length; j >= i; j--)           // ★ 从后往前挪，否则覆盖
        L->data[j] = L->data[j-1];
    L->data[i-1] = e;
    L->length++;
    return 1;
}

/* 顺序表：删除第 i 个 */
int ListDelete(SqList *L, int i, ElemType *e) {
    if (i < 1 || i > L->length) return 0;
    *e = L->data[i-1];
    for (int j = i; j < L->length; j++)            // ★ 从前往后挪
        L->data[j-1] = L->data[j];
    L->length--;
    return 1;
}

/* 有序顺序表去重，O(n) 一趟（读写双指针）*/
int Delete_Same(SqList *L) {
    if (L->length == 0) return 0;
    int k = 0;                                     // k 是写指针
    for (int i = 1; i < L->length; i++)
        if (L->data[k] != L->data[i]) L->data[++k] = L->data[i];
    L->length = k + 1;
    return 1;
}
```

```c
/* 链表：头插法建表（结果逆序）*/
LinkList CreateByHead(int a[], int n) {
    LinkList L = (LNode *)malloc(sizeof(LNode)); L->next = NULL;
    for (int i = 0; i < n; i++) {
        LNode *s = (LNode *)malloc(sizeof(LNode));
        s->data = a[i];
        s->next = L->next; L->next = s;            // 头插
    }
    return L;
}

/* 链表：尾插法建表（保序）★ 建表默认用这个 */
LinkList CreateByTail(int a[], int n) {
    LinkList L = (LNode *)malloc(sizeof(LNode)); L->next = NULL;
    LNode *r = L;                                  // r 始终指向尾结点
    for (int i = 0; i < n; i++) {
        LNode *s = (LNode *)malloc(sizeof(LNode));
        s->data = a[i]; s->next = NULL;
        r->next = s; r = s;
    }
    return L;
}

/* 链表：按序号查找第 i 个结点（i 从 1 计）*/
LNode *GetElem(LinkList L, int i) {
    if (i < 1) return NULL;
    LNode *p = L; int j = 0;
    while (p && j < i) { p = p->next; j++; }
    return p;
}

/* 链表：判断是否有环，并返回入环点（快慢指针）*/
LNode *FindLoopStart(LinkList L) {
    LNode *slow = L->next, *fast = L->next;
    while (fast && fast->next) {
        slow = slow->next; fast = fast->next->next;
        if (slow == fast) break;                   // 相遇 → 有环
    }
    if (!fast || !fast->next) return NULL;         // 无环
    LNode *p = L->next, *q = slow;                 // ★ 一个从头走，一个从相遇点走
    while (p != q) { p = p->next; q = q->next; }   //   相遇处即入环点
    return p;
}

/* 链表：按奇偶位置拆成两个表（A 保序，B 逆序）*/
void Split(LinkList A, LinkList B) {
    LNode *p = A->next, *ra = A, *q;
    B->next = NULL;
    int i = 0;
    while (p) {
        i++;
        q = p->next;                               // 存后继
        if (i % 2 == 0) { p->next = B->next; B->next = p; }   // 偶数位头插到 B
        else            { ra->next = p; ra = p; }             // 奇数位尾插到 A
        p = q;
    }
    ra->next = NULL;
}
```

**要点**
- 顺序表插入**从后往前**挪，删除**从前往后**挪。反了就覆盖。
- `GetElem` 里 `p = L; j = 0;` 从**头结点**起步计数，所以循环走 i 步后 p 正好落在第 i 个数据结点；`i < 1` 由开头的守卫直接返回 NULL。
- 判环后找入环点：**从头和从相遇点同速走，相遇即入口**（Floyd 判圈）。这是数学结论，记结论即可。

### 10.2 栈与队列（★★★）

```c
/* 括号匹配 */
int BracketCheck(char str[], int len) {
    char st[MaxSize]; int top = -1;
    for (int i = 0; i < len; i++) {
        char c = str[i];
        if (c == '(' || c == '[' || c == '{') st[++top] = c;
        else if (c == ')' || c == ']' || c == '}') {
            if (top == -1) return 0;               // 右括号多了
            char t = st[top--];
            if (c == ')' && t != '(') return 0;
            if (c == ']' && t != '[') return 0;
            if (c == '}' && t != '{') return 0;
        }
    }
    return top == -1;                              // ★ 结束时栈必须为空
}

/* 中缀转后缀（只含 + - * / 和括号）*/
int prec(char op) { return (op == '+' || op == '-') ? 1 : 2; }
void Infix2Postfix(char in[], char out[]) {
    char st[MaxSize]; int top = -1, k = 0;
    for (int i = 0; in[i]; i++) {
        char c = in[i];
        if (c >= '0' && c <= '9') out[k++] = c;                    // 操作数直接输出
        else if (c == '(') st[++top] = c;
        else if (c == ')') {
            while (top >= 0 && st[top] != '(') out[k++] = st[top--];
            top--;                                                  // 弹掉 '('
        } else {                                                    // 运算符
            while (top >= 0 && st[top] != '(' && prec(st[top]) >= prec(c))
                out[k++] = st[top--];                               // ★ 优先级不低的先出栈
            st[++top] = c;
        }
    }
    while (top >= 0) out[k++] = st[top--];
    out[k] = '\0';
}

/* 后缀表达式求值 */
int EvalPostfix(char post[]) {
    int st[MaxSize], top = -1;
    for (int i = 0; post[i]; i++) {
        char c = post[i];
        if (c >= '0' && c <= '9') st[++top] = c - '0';
        else {
            int b = st[top--], a = st[top--];                       // ★ 先弹的是右操作数
            switch (c) {
                case '+': st[++top] = a + b; break;
                case '-': st[++top] = a - b; break;
                case '*': st[++top] = a * b; break;
                case '/': st[++top] = a / b; break;
            }
        }
    }
    return st[top];
}
```

**要点**
- 中缀转后缀：**遇到运算符，先把栈里优先级 ≥ 它的都弹出**，再压自己。左括号只有遇到右括号才弹。
- 后缀求值：`b = pop(); a = pop();` 顺序不能反，`a - b` 才对。

### 10.3 树（★★★ 遍历全套 + ★★ 其余）

```c
/* 先序 非递归 */
void PreOrderNoRec(BiTree T) {
    BiTNode *st[MaxSize]; int top = -1;
    BiTNode *p = T;
    while (p || top != -1) {
        if (p) { visit(p); st[++top] = p; p = p->lchild; }   // 访问后压栈，向左走
        else   { p = st[top--]; p = p->rchild; }              // 回退，转向右
    }
}

/* 中序 非递归 */
void InOrderNoRec(BiTree T) {
    BiTNode *st[MaxSize]; int top = -1;
    BiTNode *p = T;
    while (p || top != -1) {
        if (p) { st[++top] = p; p = p->lchild; }              // 只压栈，不访问
        else   { p = st[top--]; visit(p); p = p->rchild; }    // 弹出时才访问
    }
}

/* 后序 非递归（双栈法，最好背这个，不易错）*/
void PostOrderNoRec(BiTree T) {
    if (!T) return;
    BiTNode *s1[MaxSize], *s2[MaxSize]; int t1 = -1, t2 = -1;
    s1[++t1] = T;
    while (t1 != -1) {
        BiTNode *p = s1[t1--];
        s2[++t2] = p;                                         // s2 收集：根右左
        if (p->lchild) s1[++t1] = p->lchild;
        if (p->rchild) s1[++t1] = p->rchild;
    }
    while (t2 != -1) visit(s2[t2--]);                         // 逆序输出 = 左右根
}

/* 求二叉树宽度（某层结点最多） */
int TreeWidth(BiTree T) {
    if (!T) return 0;
    BiTNode *Q[MaxSize]; int front = 0, rear = 0, maxw = 0;
    Q[rear++] = T;
    while (front < rear) {
        int size = rear - front;
        if (size > maxw) maxw = size;
        while (size--) {
            BiTNode *p = Q[front++];
            if (p->lchild) Q[rear++] = p->lchild;
            if (p->rchild) Q[rear++] = p->rchild;
        }
    }
    return maxw;
}

/* 交换所有结点的左右子树 */
void SwapChild(BiTree T) {
    if (!T) return;
    SwapChild(T->lchild); SwapChild(T->rchild);
    BiTNode *t = T->lchild; T->lchild = T->rchild; T->rchild = t;
}

/* 判断两棵树是否相同 */
int Similar(BiTree A, BiTree B) {
    if (!A && !B) return 1;
    if (!A || !B) return 0;
    return A->data == B->data && Similar(A->lchild,B->lchild) && Similar(A->rchild,B->rchild);
}

/* 由 先序 + 中序 构造二叉树 */
BiTree PreInCreate(ElemType pre[], int pl, int pr, ElemType in[], int il, int ir) {
    if (pl > pr) return NULL;
    BiTNode *root = (BiTNode *)malloc(sizeof(BiTNode));
    root->data = pre[pl];                                     // 先序第一个是根
    int k = il;
    while (in[k] != pre[pl]) k++;                             // 中序里找根，分出左右
    int llen = k - il;                                        // 左子树长度
    root->lchild = PreInCreate(pre, pl+1,      pl+llen, in, il,  k-1);
    root->rchild = PreInCreate(pre, pl+llen+1, pr,      in, k+1, ir);
    return root;
}
```

**中序线索二叉树（★★）**
```c
typedef struct ThreadNode {
    ElemType data;
    struct ThreadNode *lchild, *rchild;
    int ltag, rtag;                 // tag=0 指向孩子，tag=1 指向线索
} ThreadNode, *ThreadTree;

ThreadNode *pre_t = NULL;           // 全局：中序前驱

void InThread(ThreadTree p) {
    if (!p) return;
    InThread(p->lchild);
    if (!p->lchild) { p->lchild = pre_t; p->ltag = 1; }       // 左空 → 指前驱
    if (pre_t && !pre_t->rchild) { pre_t->rchild = p; pre_t->rtag = 1; }  // 前驱右空 → 指后继
    pre_t = p;
    InThread(p->rchild);
}
void CreateInThread(ThreadTree T) {
    pre_t = NULL;
    if (T) { InThread(T); if (pre_t->rchild == NULL) pre_t->rtag = 1; }   // 处理最后一个
}

/* 中序线索树：找 p 的中序后继 */
ThreadNode *NextNode(ThreadNode *p) {
    if (p->rtag == 1) return p->rchild;                       // 右是线索，直接是后继
    ThreadNode *q = p->rchild;                                // 否则是右子树最左下结点
    while (q->ltag == 0) q = q->lchild;
    return q;
}
```
> **口诀**：左空指前驱，右空指后继，前驱的右在下一步补。找后继＝「右是线索就是它，否则右子树一路向左」。

**哈夫曼树（★★）**
```c
typedef struct { int weight, parent, lchild, rchild; } HTNode;

void CreateHuffman(HTNode HT[], int w[], int n) {
    int m = 2*n - 1;                                          // 总结点数
    for (int i = 0; i < m; i++) { HT[i].weight = (i<n)?w[i]:0; HT[i].parent = HT[i].lchild = HT[i].rchild = -1; }
    for (int k = n; k < m; k++) {
        int s1 = -1, s2 = -1;
        for (int i = 0; i < k; i++) {                         // 找两个最小且无双亲的
            if (HT[i].parent != -1) continue;
            if (s1 == -1 || HT[i].weight < HT[s1].weight) { s2 = s1; s1 = i; }
            else if (s2 == -1 || HT[i].weight < HT[s2].weight) s2 = i;
        }
        HT[s1].parent = k; HT[s2].parent = k;
        HT[k].lchild = s1; HT[k].rchild = s2;
        HT[k].weight = HT[s1].weight + HT[s2].weight;
    }
}
```
> n 个叶子的哈夫曼树共 **2n−1** 个结点，**没有度为 1 的结点**。这两条选择题年年考。

**并查集（★★）**
```c
int ufs[MaxSize];
void UFS_Init(int n) { for (int i = 0; i < n; i++) ufs[i] = -1; }   // -1 表示自己是根

int Find(int x) {                                              // 带路径压缩
    if (ufs[x] < 0) return x;
    return ufs[x] = Find(ufs[x]);
}
void Union(int a, int b) {                                     // 按秩（结点数）合并
    int ra = Find(a), rb = Find(b);
    if (ra == rb) return;
    if (ufs[ra] > ufs[rb]) { int t = ra; ra = rb; rb = t; }     // ufs 存负的结点数，小的是大树
    ufs[ra] += ufs[rb];
    ufs[rb] = ra;
}
```

```c
/* 邻接矩阵 → 邻接表 */
void Mat2List(MGraph G, ALGraph *AL) {
    AL->vexnum = G.numVertices; AL->arcnum = G.numEdges;
    for (int i = 0; i < G.numVertices; i++) AL->adjlist[i].first = NULL;
    for (int i = 0; i < G.numVertices; i++)
        for (int j = G.numVertices - 1; j >= 0; j--)           // ★ 倒序头插，链表才是升序
            if (G.Edge[i][j]) {
                ArcNode *s = (ArcNode *)malloc(sizeof(ArcNode));
                s->adjvex = j;
                s->next = AL->adjlist[i].first;
                AL->adjlist[i].first = s;
            }
}

/* BFS 求无权图单源最短路径（层数即距离）*/
void BFS_MinDist(MGraph G, int u, int d[]) {
    for (int i = 0; i < G.numVertices; i++) { d[i] = -1; visited[i] = 0; }
    int Q[MaxSize], f = 0, r = 0;
    d[u] = 0; visited[u] = 1; Q[r++] = u;
    while (f < r) {
        int v = Q[f++];
        for (int w = 0; w < G.numVertices; w++)
            if (G.Edge[v][w] && !visited[w]) {
                d[w] = d[v] + 1;                                // ★ 距离 = 上一层 + 1
                visited[w] = 1; Q[r++] = w;
            }
    }
}

/* 拓扑排序（Kahn，输出一个合法序列）*/
int TopoSort(MGraph G, int order[]) {
    int n = G.numVertices, indeg[MAXV], st[MAXV], top = -1, cnt = 0;
    for (int i = 0; i < n; i++) indeg[i] = 0;
    for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) if (G.Edge[i][j]) indeg[j]++;
    for (int i = 0; i < n; i++) if (indeg[i] == 0) st[++top] = i;
    while (top != -1) {
        int v = st[top--];
        order[cnt++] = v;
        for (int w = 0; w < n; w++)
            if (G.Edge[v][w] && --indeg[w] == 0) st[++top] = w;
    }
    return cnt == n;                                            // 0 表示有环
}

/* Prim：从 v0 出发求最小生成树权值和 */
int Prim(MGraph G, int v0) {
    int n = G.numVertices, lowcost[MAXV], inMST[MAXV], sum = 0;
    for (int i = 0; i < n; i++) { lowcost[i] = G.Edge[v0][i] ? G.Edge[v0][i] : INF; inMST[i] = 0; }
    inMST[v0] = 1; lowcost[v0] = 0;
    for (int k = 1; k < n; k++) {
        int min = INF, v = -1;
        for (int i = 0; i < n; i++)                             // 找 lowcost 最小的树外点
            if (!inMST[i] && lowcost[i] < min) { min = lowcost[i]; v = i; }
        if (v == -1) return -1;                                 // 非连通
        inMST[v] = 1; sum += min;
        for (int i = 0; i < n; i++) {                           // ★ 用新点更新 lowcost
            int w = G.Edge[v][i] ? G.Edge[v][i] : INF;
            if (!inMST[i] && w < lowcost[i]) lowcost[i] = w;
        }
    }
    return sum;
}

/* Dijkstra：求 v0 到各点最短路 */
void Dijkstra(MGraph G, int v0, int dist[], int path[]) {
    int n = G.numVertices, final[MAXV];
    for (int i = 0; i < n; i++) {
        dist[i] = G.Edge[v0][i] ? G.Edge[v0][i] : INF;
        path[i] = (dist[i] < INF) ? v0 : -1;
        final[i] = 0;
    }
    dist[v0] = 0; final[v0] = 1; path[v0] = -1;
    for (int k = 1; k < n; k++) {
        int min = INF, v = -1;
        for (int i = 0; i < n; i++)
            if (!final[i] && dist[i] < min) { min = dist[i]; v = i; }
        if (v == -1) break;
        final[v] = 1;
        for (int i = 0; i < n; i++) {                           // ★ 松弛
            int w = G.Edge[v][i] ? G.Edge[v][i] : INF;
            if (!final[i] && dist[v] + w < dist[i]) {
                dist[i] = dist[v] + w;
                path[i] = v;
            }
        }
    }
}

/* Floyd：所有点对最短路，三重循环，k 必须在最外层 */
void Floyd(MGraph G, int A[][MAXV]) {
    int n = G.numVertices;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            A[i][j] = (i == j) ? 0 : (G.Edge[i][j] ? G.Edge[i][j] : INF);
    for (int k = 0; k < n; k++)                                 // ★ k 在最外层，这是 Floyd 的灵魂
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (A[i][k] + A[k][j] < A[i][j]) A[i][j] = A[i][k] + A[k][j];
}
```

**要点**
- **Prim vs Dijkstra 只差一行**：Prim 更新 `lowcost[i] = w`（到**树**的距离），Dijkstra 更新 `dist[i] = dist[v] + w`（到**源点**的距离）。看清楚这一行就不会混。
- **Floyd 的 k 必须在最外层**，写成内层就是错的。理解：k 表示「允许经过前 k 个点中转」。
- 拓扑排序用栈或队列都行，用**栈**得到的序列和用**队列**通常不同——题目问「可能的拓扑序列」就都合法。

**关键路径 AOE（★★，代码不考但要会算）**
1. 求 **ve**（事件最早发生时间）：按**拓扑序**正推，`ve[j] = max(ve[i] + w(i,j))`
2. 求 **vl**（事件最迟发生时间）：按**逆拓扑序**倒推，`vl[i] = min(vl[j] - w(i,j))`
3. 求 **e**（活动最早开始）= 起点的 ve；**l**（活动最迟开始）= 终点的 vl − 权
4. **e == l 的活动就是关键活动**，串起来就是关键路径

### 10.5 查找（★★★ BST / ★★ AVL、散列）

```c
/* BST 查找（非递归）*/
BiTNode *BST_Search(BiTree T, ElemType key) {
    while (T && T->data != key) T = (key < T->data) ? T->lchild : T->rchild;
    return T;
}

/* BST 插入（递归）*/
int BST_Insert(BiTree *T, ElemType k) {
    if (*T == NULL) {
        *T = (BiTNode *)malloc(sizeof(BiTNode));
        (*T)->data = k; (*T)->lchild = (*T)->rchild = NULL;
        return 1;
    }
    if (k == (*T)->data) return 0;                              // 已存在，不插
    if (k <  (*T)->data) return BST_Insert(&(*T)->lchild, k);
    return BST_Insert(&(*T)->rchild, k);
}

/* BST 删除 —— 三种情况 */
BiTree BST_Delete(BiTree T, ElemType k) {
    if (!T) return NULL;
    if (k < T->data)      T->lchild = BST_Delete(T->lchild, k);
    else if (k > T->data) T->rchild = BST_Delete(T->rchild, k);
    else {
        if (!T->lchild)  { BiTNode *r = T->rchild; free(T); return r; }   // ① 无左孩子
        if (!T->rchild)  { BiTNode *r = T->lchild; free(T); return r; }   // ② 无右孩子
        BiTNode *s = T->rchild;                                           // ③ 双孩子：
        while (s->lchild) s = s->lchild;                                  //   找右子树最小（中序后继）
        T->data = s->data;                                                //   值搬上来
        T->rchild = BST_Delete(T->rchild, s->data);                       //   在右子树里删掉它
    }
    return T;
}
```

**AVL 四种旋转（★★，代码能读即可，手算用三结点重构法）**

| 情形 | 失衡形状 | 怎么转 |
|---|---|---|
| LL | 左孩子的左子树过高 | 右单旋 |
| RR | 右孩子的右子树过高 | 左单旋 |
| LR | 左孩子的右子树过高 | 先对左孩子左旋，再对失衡点右旋 |
| RL | 右孩子的左子树过高 | 先对右孩子右旋，再对失衡点左旋 |

> **考场别背这张表**，用 §1.1 的三结点重构法：找失衡点 A → A 的高子树根 B → B 的高子树根 C，把 A/B/C 按大小掰成正三角，剩下 4 棵子树按序挂回。结果和查表旋转完全一致。

**散列表（★★，主要考手算 ASL）**
```c
/* 除留余数 + 线性探测 */
int Hash_Insert(int HT[], int m, int key) {          // HT 初值全为 -1
    int h = key % m;
    for (int i = 0; i < m; i++) {
        int pos = (h + i) % m;                       // 线性探测：+1 +2 +3...
        if (HT[pos] == -1) { HT[pos] = key; return pos; }
    }
    return -1;                                        // 表满
}
```
- **平方探测**：`pos = (h ± i²) % m`，表长 m 必须是形如 4k+3 的素数
- **ASL 成功** = Σ(每个关键字的比较次数) / **关键字个数**
- **ASL 失败** = Σ(每个散列地址上查找失败的比较次数) / **散列函数值域大小**
> 2010 年 41 题就栽在这：`ASL失败` 的分母是 `mod 7` 的 7，不是表长 10。

### 10.6 排序（★★★ 全部默写）

```c
/* 直接插入排序 A[0..n-1] */
void InsertSort(int A[], int n) {
    for (int i = 1; i < n; i++) {
        if (A[i] < A[i-1]) {
            int tmp = A[i], j;
            for (j = i - 1; j >= 0 && A[j] > tmp; j--) A[j+1] = A[j];   // 边比边挪
            A[j+1] = tmp;
        }
    }
}

/* 折半插入排序：只减少比较次数，移动次数不变 */
void BInsertSort(int A[], int n) {
    for (int i = 1; i < n; i++) {
        int tmp = A[i], low = 0, high = i - 1;
        while (low <= high) {
            int mid = (low + high) / 2;
            if (A[mid] > tmp) high = mid - 1;
            else              low  = mid + 1;        // ★ 相等时往右，保证稳定
        }
        for (int j = i - 1; j >= low; j--) A[j+1] = A[j];
        A[low] = tmp;
    }
}

/* 希尔排序 */
void ShellSort(int A[], int n) {
    for (int gap = n / 2; gap >= 1; gap /= 2)
        for (int i = gap; i < n; i++) {
            int tmp = A[i], j;
            for (j = i - gap; j >= 0 && A[j] > tmp; j -= gap) A[j+gap] = A[j];
            A[j+gap] = tmp;
        }
}

/* 冒泡排序（带提前退出）*/
void BubbleSort(int A[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int flag = 0;
        for (int j = 0; j < n - 1 - i; j++)
            if (A[j] > A[j+1]) { int t = A[j]; A[j] = A[j+1]; A[j+1] = t; flag = 1; }
        if (!flag) return;                            // ★ 一趟没交换 → 已有序
    }
}

/* 简单选择排序 */
void SelectSort(int A[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++) if (A[j] < A[min]) min = j;
        if (min != i) { int t = A[i]; A[i] = A[min]; A[min] = t; }
    }
}

/* 二路归并排序（递归）*/
void Merge(int A[], int low, int mid, int high, int B[]) {
    for (int k = low; k <= high; k++) B[k] = A[k];
    int i = low, j = mid + 1, k = low;
    while (i <= mid && j <= high) A[k++] = (B[i] <= B[j]) ? B[i++] : B[j++];   // <= 保稳定
    while (i <= mid)  A[k++] = B[i++];
    while (j <= high) A[k++] = B[j++];
}
void MergeSort(int A[], int low, int high, int B[]) {
    if (low >= high) return;
    int mid = (low + high) / 2;
    MergeSort(A, low, mid, B);
    MergeSort(A, mid + 1, high, B);
    Merge(A, low, mid, high, B);
}

/* 基数排序（LSD，非负整数）*/
void RadixSort(int A[], int n) {
    int *tmp = (int *)malloc(sizeof(int) * n), cnt[10];
    int max = A[0];
    for (int i = 1; i < n; i++) if (A[i] > max) max = A[i];
    for (int exp = 1; max / exp > 0; exp *= 10) {
        for (int i = 0; i < 10; i++) cnt[i] = 0;
        for (int i = 0; i < n; i++) cnt[(A[i]/exp) % 10]++;
        for (int i = 1; i < 10; i++) cnt[i] += cnt[i-1];        // 前缀和 → 每个桶的末位置
        for (int i = n - 1; i >= 0; i--)                        // ★ 倒着放才稳定
            tmp[--cnt[(A[i]/exp) % 10]] = A[i];
        for (int i = 0; i < n; i++) A[i] = tmp[i];
    }
    free(tmp);
}
```

**排序性质速记（选择题高频）**

| 性质 | 算法 |
|---|---|
| **稳定**的 | 直接插入、折半插入、冒泡、归并、基数、计数 |
| **不稳定**的 | 希尔、快排、简单选择、堆排 |
| 最坏 O(n²) | 直接插入、冒泡、简单选择、**快排** |
| 恒定 O(n log n) | 归并、堆排 |
| 空间 O(n) | 归并 |
| 空间 O(log n) | 快排（递归栈） |
| 空间 O(1) | 插入、冒泡、选择、希尔、堆排 |
| **一趟后至少一个元素到最终位置** | 冒泡、快排、简单选择、堆排 |
| 与初始序列**无关**（比较次数固定） | 简单选择、折半插入（比较次数）、归并、堆排 |

> **口诀**：「**快些**（快排、希尔）**选**（简单选择）**堆**（堆排）**不稳定**」，其余都稳定。

**外部排序（★★，2012·41 和 2023·42 考过）**
- **多路归并**：k 路归并，m 个初始归并段，归并趟数 = ⌈log_k m⌉。增大 k 减少 I/O，但内部比较增加 → 用**败者树**降到 ⌈log₂k⌉
- **置换-选择排序**：用大小为 m 的内存工作区，可以生成**长度超过 m** 的归并段。最大可能长度 = n（整个文件有序时），最小 = m
- **最佳归并树**：把各归并段长度当权值建**哈夫曼树**（k 路就建 k 叉哈夫曼树），带权路径长度最小 = I/O 次数最少。段数不足时要**补长度为 0 的虚段**

### 11.3 递归三部曲（树题 90% 靠它）

老师在树的题型总结（P25）里给的原话：

> **不要去想它具体是怎么一步步实现得到结果的，要想「怎么把问题拆分成很多个子问题的结果」。**
> 觉得递归难，是因为想错了方向。

```
第一步：确定功能
    题目给了函数原型和参数 → 直接用
    没给 → 从长题干里找出「哪一句话是我们要求的东西」

第二步：确定递归结束条件
    在二叉树里就是「特殊结点」：
      · 只涉及空结点 → 写 if (!T) return 边界值;
      · 涉及叶结点   → 空结点和叶结点两个都要先写出来
    能直接 return 的就 return，不能的放到第三步

第三步：确定递推关系
    以当前结点为根的子树的结果，和它左、右两棵子树的结果，怎么算出最终结果
    想清楚这个，return 语句自然就写出来了
```

**反面示范**：一上来就想「我这个 return 到底要怎么写」——这是最常见的卡壳原因。**先做第二步，再做第三步，return 是第三步的产物。**

### 11.4 换存储结构时，递归要改哪几处

老师特别强调：**题型五（存储结构转换）的意义就是「同一个问题换个壳再考你一遍」**，所以不能背代码，要知道改哪里。

以「求叶子结点个数」为例，从二叉链表换成**孩子兄弟表示法**：

```c
typedef struct CSNode {
    ElemType data;
    struct CSNode *firstchild, *nextsibling;   // 左：第一个孩子；右：下一个兄弟
} CSNode, *CSTree;

int CountLeaf_CS(CSTree T) {
    if (T == NULL) return 0;
    if (T->firstchild == NULL)                              // 改动①：叶结点判定
        return 1 + CountLeaf_CS(T->nextsibling);            // 改动②：自己是叶子，但右兄弟还有任务
    return CountLeaf_CS(T->firstchild) + CountLeaf_CS(T->nextsibling);
}

int Height_CS(CSTree T) {
    if (T == NULL) return 0;
    int hc = Height_CS(T->firstchild) + 1;    // ★ 孩子：要 +1（下一层）
    int hs = Height_CS(T->nextsibling);       // ★ 兄弟：不 +1（同一层）
    return hc > hs ? hc : hs;
}
```

**三处改动，对照二叉链表版本看：**

| | 二叉链表 | 孩子兄弟表示法 |
|---|---|---|
| 叶结点判定 | `!T->lchild && !T->rchild` | `!T->firstchild`（**只看左**） |
| 叶结点 return | `return 1;` | `return 1 + f(T->nextsibling);`（右兄弟还没算） |
| 求高度 | `1 + max(左, 右)` | `max(1 + f(孩子), f(兄弟))`（**兄弟不加 1**） |

> 记住原因就不用背：孩子兄弟表示法里，**右指针是"同辈"不是"下一层"**，所以凡是和「层数/深度」有关的，右边一律不加 1。

### 11.5 顺序表删除的两种写法（都要会，考场任选）

老师在 P8 明说这两种「灵活应用的话在各种顺序表题目中都有帮助」。

**写法一：count 计数法**——记录「前面有几个要删的」，非目标元素往前挪那么多格
```c
void Del_X_1(SqList *L, ElemType x) {
    int count = 0;                                  // 到目前为止遇到了几个 x
    for (int i = 0; i < L->length; i++) {
        if (L->data[i] == x) count++;
        else L->data[i - count] = L->data[i];       // ★ 往前挪 count 格
    }
    L->length -= count;                             // ★ 别忘了改长度
}
```

**写法二：k 写指针法**——k 记录「已经放好了几个非目标元素」
```c
void Del_X_2(SqList *L, ElemType x) {
    int k = 0;                                      // k 是下一个非 x 元素该放的位置
    for (int i = 0; i < L->length; i++)
        if (L->data[i] != x) L->data[k++] = L->data[i];
    L->length = k;                                  // ★ 注意这里是「等于」不是「减去」
}
```

**长度处理的区别是最易错的点**：写法一 `length -= count`，写法二 `length = k`。写反了直接错。

**删除区间 [x, y] 内的元素**——两种写法都**只改一个条件**：
```c
if (L->data[i] >= x && L->data[i] <= y) count++;     // 写法一
if (L->data[i] < x || L->data[i] > y) L->data[k++] = L->data[i];   // 写法二
```

> 老师提醒：如果直接套「基本操作的删除」（找到一个就把后面全体前移），有多个 x 时是 **O(n²)**，不是 O(n)。题目没要求复杂度可以这么写，要求了就必须用上面两种。

### 11.6 链表题的心法

老师在 P15 的原话：

> 做链表的题目，无论哪种题型，核心都是**对结点指针的处理**。技巧不多，只能不断练习。
> **心里要有一张链表，把所有结点都放在心上——不要让任何一个结点断链或丢失。**

三条硬规则：
1. **删除 p 之前，必须先拿到 p 的前驱**。无论删的是「值为 x 的」还是「最小值结点」，都一样。
2. **删除之后，指针接下来该指向哪里**——如果后面还有要删的，`pre` 不能前进。
3. **改任何指针之前，先把后继存到临时变量 r**。

**链表五个题型里最重要的是快慢指针**，老师给的理由很实在：

> 它大概率**没有办法通过其他方式来解**。想不到就只能花很多时间和空间去凑。掌握了它，遇到其他链表题也都会有思路。

**头插法逆置的两个必查点**（老师逐条点名）：
- 插入操作是 `current->next = L->next; L->next = current;`，两句顺序不能反
- 操作**之前**取出 `current->next` 存好，操作**之后**把它赋回 `current`——这两步才构成循环

**哪些排序算法能用链表实现**：简单选择、直接插入、冒泡 ✅ ｜ 折半插入、希尔、快排、堆排 ❌（都要随机访问）
> 老师建议：学排序时把每个算法都试着用链表写一遍，能用的写，不能用的想清楚为什么不能。

### 11.7 图：先保住基本分，再谈进阶

老师在 P32 给的**明确优先级**：

| 优先级 | 内容 | 老师原话 |
|---|---|---|
| **必拿基本分** | ① 存储结构 ② 求度 ③ DFS + BFS | 「一定要像背诗一样就能写出来」 |
| **★★ 第二重要** | 路径问题（DFS + 回溯的 5 个变体） | 「虽然不是必考，但考的话也就考到这个难度」 |
| **★★** | 连通性、是否有环、是否是树 | 「非常经典和有用」 |
| **考得变态才会到** | Prim / Kruskal / Dijkstra / Floyd | 「重要的是知道过程和思想」 |

**判断图是否连通**——调一次 DFS，然后看 visited 是不是全 true
```c
int Connected(MGraph G) {
    for (int i = 0; i < G.numVertices; i++) visited[i] = 0;
    DFS(G, 0);                                          // ★ 只调用一次
    for (int i = 0; i < G.numVertices; i++)
        if (!visited[i]) return 0;                      // 有点没访问到 → 不连通
    return 1;
}
```
> 原理：`DFSTraverse` 里那个外层 for 循环调用 DFS 几次，就有几个连通分量。**只调一次就能全访问到 = 连通图。**

**判断无向图是否有环**——DFS 传入父结点，遇到「已访问 且 不是父结点」的邻接点就是有环
```c
int HasCycle(MGraph G, int v, int parent) {
    visited[v] = 1;
    for (int u = 0; u < G.numVertices; u++) {
        if (!G.Edge[v][u]) continue;                    // ★ 注意：这里不加 !visited[u]
        if (!visited[u]) {
            if (HasCycle(G, u, v)) return 1;            // ★ 子调用返回真要往上传
        } else if (u != parent) {
            return 1;                                   // 已访问 且 不是来的那个点 → 有环
        }
    }
    return 0;
}
```
> 和普通 DFS 的唯一区别：**邻接点的 for 里不能提前用 `!visited[u]` 过滤**，因为「已访问的邻接点」正是我们要判断的对象。

**判断图是不是树** = 无环 ✚ 连通。老师把这题留成作业——就是把上面两个函数合起来。

**有向图判环**：拓扑排序后 `count != 顶点数` 即有环（代码长，除非本来就考拓扑排序，否则用 DFS 法）。

### 11.8 复杂度阶梯：想不到最优解怎么办

老师在 P4 的原话，几乎是逐字的答题策略：

> 不要太过于追求完美的解、最优解。先训练自己**把题目解出来**，这是目的；再去训练把解答变得更优。
> **如果你追求不了完美，或者最终没把问题解决，那你的分数还不如用暴力解来得高。**

配合真题评分细则（2016·43 原文）：

```
        O(n)  ██████████████ 13 分
   O(n log n) ███████████    11 分
       O(n²)  █████████       9 分
      更差    ███████         7 分
       空着   ·               0 分
```

**考场决策树：**
```
读完题 → 5 分钟内想到最优解？
  ├─ 是 → 写最优解
  └─ 否 → 先把暴力解完整写出来（含算法思想 + 注释）
           ├─ 还有时间 → 在旁边补一句「可优化为 O(?)，思路是……」
           └─ 没时间   → 收工，去做下一题
```

**绝对不要**：在草稿纸上死磕最优解到最后一分钟，交白卷。

### 11.9 哪些内容可以战略性放弃

老师在讲解中明确说过「大概率不考 / 可以放过」的（**你自己权衡，不是我的判断**）：

| 内容 | 老师原话 | 建议 |
|---|---|---|
| 线索二叉树的线索化代码 | 「大概率不会考，只是做个后备选择，万一考了呢」 | 会读会改即可，别花时间默写 |

| 非递归遍历 | 「能掌握递归就掌握递归，不要为了掌握非递归把递归也忘了」 | 递归优先，非递归后序用双栈法（最不易错） |
| B 树 / B+ 树插入删除代码 | 考纲只要求理解 | **只练手算画图**，写代码零收益 |

**反过来，老师点名「没考过但极大概率会考」的**：
- **归并排序的 Merge 函数**——「这个东西是重点加基础，它虽然没有考过，但是有非常大概率会考到，大家一定要熟练掌握」

按这个顺序扫一遍，比通读代码快得多：

1. **算法思想写了吗**（4~6 分，最贵的一项）
2. **返回值类型和参数**对不对？参数里该加 `*` 的加了吗（要改的顺序表/树根必须传指针或指针的指针）
3. **循环边界**：`<` 还是 `<=`？下标从 0 还是 1？
4. **长度/计数**改了吗（顺序表的 `length`、树的 `count`）
5. **链表**：每个 `p->next` 前，p 可能是 NULL 吗？删除后 pre 动了吗？
6. **递归**：出口写了吗？
7. **题目要求复杂度吗**——要求了才写

## 12. 逐专题精华

### 12.1 顺序表（P4–P9）

**真题分布**：09–25 共 **7 道**在顺序表上。【老师原话】「为什么顺序表占这么多？因为顺序表就是一维数组，我们后面学的查找排序这些算法大部分都是针对一维数组的。」

#### 查找的三分法（题型一）

| 题干条件 | 用什么 |
|---|---|
| 没说有序 | **顺序查找**（for + if） |
| 明确说「有序顺序表」 | **折半查找** |
| 求最大/最小值 | **简单选择排序的内层循环**（推荐）或冒泡 |

【老师原话】为什么找最值选简单选择：直接插入不行（一趟定不下位置），归并不行，堆可以但代码长，快排属于双指针题型 —— **只有简单选择和冒泡是「一趟就能定死一个最值」的**。

```c
/* 找最小值：就是简单选择排序抽掉外层循环 */
ElemType FindMin(SqList *L) {
    int min = 0;
    for (int i = 1; i < L->length; i++)      /* ★ i 从 1 开始，i=0 和 min=0 比较没意义 */
        if (L->data[i] < L->data[min]) min = i;
    return L->data[min];
}
```

**返回值的坑**：顺序表逻辑位序从 1 开始，数组下标从 0 开始。查找位置的题要 `return i + 1;`

#### for 还是 while？（一个很实用的判断标准）

【老师原话】循环变量**每次都固定变化**（`i++`）→ 用 `for`；变化**取决于 if 判断**（折半的 low/high、双指针的 low/high）→ 用 `while`。

#### 合并（题型三）

【老师原话】「归并的 Merge 函数**没考过，但有非常大概率会考**，教材第二章线性表的应用里明确讲了合并两个线性表。一定要熟练掌握。」

**变体：两个递增表合并成一个递减表** —— 两种做法
1. 先按递增合并，再整体逆置
2. **从后往前归并**（推荐，一趟搞定）：i、j 从末尾往前，k 从 0 往后，每次取**大**的

```c
void MergeDesc(int A[], int m, int B[], int n, int C[]) {
    int i = m - 1, j = n - 1, k = 0;         /* ★ i、j 从末尾，k 从 0 */
    while (i >= 0 && j >= 0)                 /* ★ 条件是 >= 0，不是 < length */
        C[k++] = (A[i] > B[j]) ? A[i--] : B[j--];   /* ★ 取大的，指针 -- */
    while (i >= 0) C[k++] = A[i--];
    while (j >= 0) C[k++] = B[j--];
}
```
【老师原话】「这三处红色的地方（条件 `>=0`、变化 `--`、取大的）大家一定不要写习惯了就把这里写错。」

#### 双指针（题型四）—— 两大类

**第一类：逆置类**。关键字 = 循环左移 / 右移 / 逆置 / 互换位置。全部用 `Reverse` + 组合逆置。

> **左移和右移的三次逆置顺序不同**（老师专门澄清过）：
> - 循环**左移** k 位：`Reverse(0,n-1)` → 前 `n-k` 位逆置 → 后 `k` 位逆置
> - 循环**右移** k 位：`Reverse(0,n-1)` → 前 `k` 位逆置 → 后 `n-k` 位逆置
> - 等价做法（王道版，左移 p 位）：`Reverse(0,p-1); Reverse(p,n-1); Reverse(0,n-1);`
>
> 两条路都对，**记一条就够，但要能验算**：拿 `12345` 左移 2 位应得 `34512`，现场推一遍。

**第二类：快排 partition 类**。关键字 = 划分 / 分成两部分 / 奇偶分离 / 前面都比 x 小。

```c
/* 奇数全部移到偶数前面 —— partition 只改两个条件 */
void EvenOdd(SqList *L) {
    if (L->length == 0) return;
    int low = 0, high = L->length - 1;
    ElemType temp = L->data[low];                           /* ★ 挖出第一个，该位置可被覆盖 */
    while (low < high) {
        while (low < high && L->data[high] % 2 == 0) high--; /* ← 改这里：右边找奇数 */
        L->data[low] = L->data[high];
        while (low < high && L->data[low] % 2 != 0) low++;   /* ← 改这里：左边找偶数 */
        L->data[high] = L->data[low];
    }
    L->data[low] = temp;                                     /* ★ 相遇处放回 */
}
```
> 和标准 partition 唯一的区别就是那**两行 while 的条件**。基准值换成 `temp`（挖出的第一个元素），最后放回相遇点。

#### 三个等长递增序列求公共元素（三指针）

```c
int MaxOf3(int a, int b, int c) { int m = a > b ? a : b; return m > c ? m : c; }

int FindCommon(int A[], int B[], int C[], int n, int out[]) {
    int i = 0, j = 0, k = 0, cnt = 0;
    while (i < n && j < n && k < n) {
        if (A[i] == B[j] && B[j] == C[k]) { out[cnt++] = A[i]; i++; j++; k++; }
        else {
            int mx = MaxOf3(A[i], B[j], C[k]);
            if (A[i] < mx) i++;                  /* ★ 非最大者全部前进 */
            if (B[j] < mx) j++;
            if (C[k] < mx) k++;
        }
    }
    return cnt;
}
```
> **规则：除最大值以外的指针全部后移。** 和 2020 年三元组最小距离是姊妹题（那题是「最小者前进」，因为目标相反）。

#### 删除最小值，空位由末元素填补（经典题）

```c
int DelMin(SqList *L, int *ok) {
    if (L->length == 0) { *ok = 0; printf("顺序表为空\n"); return -1; }  /* 题目要求输出错误信息 */
    int min = 0;
    for (int i = 1; i < L->length; i++) if (L->data[i] < L->data[min]) min = i;
    int val = L->data[min];
    L->data[min] = L->data[L->length - 1];    /* ★ 末元素填补，不用整体前移 */
    L->length--;
    *ok = 1;
    return val;
}
```
【老师原话】「题目说『空出的位置由最后一个元素填补』，那它就**不是**我们理解的那种平移删除。**题目的每一个要求都可能是踩分点，一定不要忽略。**」

### 12.2 链表（P10–P16）

#### 第一原则：用原结点

【老师原话】「尽量能用原本的结点就用原本的结点，**不要新建结点把 data 复制过去**。也有同学把链表全部复制到数组上再操作，这种办法不是不行，但它**绝不是最佳**。想真正提高链表能力，就不要用这些投机取巧、占用过多辅助空间的办法。」

#### 链表的简单选择排序（题型一）

顺序表版是「找最小 → 交换」；**链表版是「找最小 → 从无序区摘下 → 尾插到有序区」**。

```c
void SelectSortList(LinkList h) {
    LNode *r = h;                       /* r 指向有序区尾 */
    LNode *q = h->next;                 /* q 指向无序区头 */
    h->next = NULL;
    while (q != NULL) {
        LNode *pre = NULL, *p = q;      /* p 遍历无序区，pre 是 p 的前驱 */
        LNode *premin = NULL, *min = q; /* min 是最小结点，premin 是它的前驱 */
        while (p != NULL) {
            if (p->data < min->data) { min = p; premin = pre; }
            pre = p; p = p->next;
        }
        if (min == q) q = q->next;                  /* ★ 最小结点就是无序区头，先移动 q */
        else premin->next = min->next;              /*    否则正常摘下 */
        r->next = min; r = min; min->next = NULL;   /* 尾插到有序区 */
    }
}
```
> **最容易漏的一句是 `if (min == q) q = q->next;`** —— 如果要删的正好是无序区的头，不先移动 q，q 就成了野指针。老师专门强调了这个特殊情况。

#### 不改动链表，按递增顺序输出（题型一的简化版）

不删不插，只比较和输出。用两个**值变量**代替指针操作：

```c
int PrintAsc(LinkList h, int out[]) {
    int n = 0, pre = INT_MIN;                     /* pre = 上一轮选出的最小值 */
    for (LNode *c = h->next; c != NULL; c = c->next) {    /* 外层只负责控制轮数 */
        int min = INT_MAX, found = 0;
        for (LNode *p = h->next; p != NULL; p = p->next)
            if (p->data < min && p->data > pre)   /* ★ 比 min 小，且比上一轮的大 */
                { min = p->data; found = 1; }
        if (!found) break;
        out[n++] = min;
        pre = min;                                 /* ★ 下一轮只找比它大的 */
    }
    return n;
}
```
> **`pre` 就是「已输出集合」的替身**：不用标记也不用删除，靠「必须比上一轮大」把已输出的全部筛掉。

#### 头插法逆置（题型二）

【老师原话】「逆置**单独考的可能性不大，但它一定会结合其他考法**。比如两个递增合并成一个递减 —— 就是把尾插换成头插。所以头插法要能**闭着眼睛写出来**。」

两个必查点：
1. 插入是 `current->next = L->next; L->next = current;`，**顺序不能反**
2. 操作**前**存 `next`，操作**后**赋回 `current` —— 这两步才构成循环

#### 合并与拆分（题型三）

**链表合并比顺序表方便的地方**：顺序表要用两个 while 逐个复制剩余元素，**链表只需要一个 if，直接把整条挂上去**：
```c
r->next = pa ? pa : pb;      /* ★ 一句话接管剩余的整条链 */
```
- 递增 → 递增：**尾插**（要维护 r 尾指针）
- 递增 → 递减：**头插**（不需要 r）

**就地拆分**：C = (a₁,b₁,a₂,b₂,…) → A 正序、B 逆序
```c
void SplitAB(LinkList C, LinkList A, LinkList B) {
    LNode *p = C->next, *r = A, *nxt;
    A->next = NULL; B->next = NULL;
    int cnt = 0;
    while (p != NULL) {
        cnt++;
        nxt = p->next;                                              /* ★ 先存后继 */
        if (cnt % 2 == 1) { r->next = p; r = p; p->next = NULL; }   /* 奇数位尾插 A */
        else              { p->next = B->next; B->next = p; }        /* 偶数位头插 B */
        p = nxt;
    }
}
```
> **正序用尾插、逆序用头插** —— 这是拆分题的全部秘密。

#### 快慢指针（题型四，最重要）

【老师原话】「五个题型里最重要的是快慢指针，因为它**大概率没有办法通过其他方式来解**。」

**三个用途**：① 判环 ② 找指定位置结点（中点 / 倒数第 k 个）③ 找公共结点（2012 真题）

**「快」不一定是走两步**：
| 题目 | 快指针怎么快 |
|---|---|
| 判环、找中点 | 走两步 |
| 倒数第 k 个 | **先走 k 步**，然后同速 |
| 两链表公共后缀 | **先走 \|m−n\| 步**，然后同速 |

**判环 + 找中点用同一个模板**，初值都是 `slow = fast = head`（头结点）：
```c
while (fast != NULL && fast->next != NULL) {    /* ★ 两个条件都要，否则 fast->next->next 会崩 */
    slow = slow->next;
    fast = fast->next->next;
    if (slow == fast) return 1;                  /* 判环时加这句 */
}
/* 找中点时，循环结束后 slow 就是中间结点（偶数个取前面那个） */
```

#### 删除最小值结点 —— 省两个指针的写法

```c
void DelMinNode(LinkList h) {
    if (h->next == NULL) return;
    LNode *p = h, *min = h;             /* ★ p->next 是当前比较的结点，min->next 是最小结点 */
    while (p->next != NULL) {           /* ★ 条件是 p->next，不是 p */
        if (p->next->data < min->next->data) min = p;
        p = p->next;
    }
    LNode *t = min->next;
    min->next = t->next;                /* 直接就能删，不需要额外的 premin */
    free(t);
}
```
> 让指针指向**前驱**，用 `p->next` 表示实际结点 —— 四个指针缩成两个。

#### 双向链表改造（真题风格的"看起来很乱"的题）

题：rlink 保持原次序，用 llink 把所有结点按值从小到大链起来。

【老师原话】「大家**根本不需要把改造后的链表画出来**，因为它会非常混乱。你只需要知道：**在原链表中找下一个结点用 rlink，在有序表中走用 llink** —— 这一点就够了。」

> 这是应对「读不懂题、画不出图」类真题的通用心法：**分清哪个指针域负责哪件事，然后只盯着自己要做的那件事。**

### 12.3 栈和队列（P17–P18）

【老师原话】「栈和队列的**真题考评是空白的 —— 从来没有考过代码题**。大家主要还是把重点和练习时间多分配给顺序表和链表。」

#### 考场书写规范（能省大量篇幅）

【老师原话】「如果题目需要用栈，**直接写 `Stack S; InitStack(S); Push(S,x); Pop(S,x);` 就可以，不需要去实现栈的结构体和基本操作**。严蔚敏教材里的伪代码也是这么写的，后面加个注释说明 S 是栈就行。老师能知道你的用意。」

同理，队列可以直接用**数组 + front/rear 两个指针**，不必定义结构体。

#### 共享栈（判满条件取决于初值和入栈方式）

```c
/* 约定：top1 从 0 向右，top2 从 MaxSize-1 向左，都是「先赋值再移动」 */
int ShFull(ShStack *S) { return S->top1 - 1 == S->top2; }
int ShPush1(ShStack *S, ElemType x) { if (ShFull(S)) return 0; S->data[S->top1++] = x; return 1; }
int ShPush2(ShStack *S, ElemType x) { if (ShFull(S)) return 0; S->data[S->top2--] = x; return 1; }
int ShPop1(ShStack *S, ElemType *x) { if (S->top1 == 0)          return 0; *x = S->data[--S->top1]; return 1; }
int ShPop2(ShStack *S, ElemType *x) { if (S->top2 == MaxSize - 1) return 0; *x = S->data[++S->top2]; return 1; }
```

**判满条件对照表**（这是共享栈唯一的考点）：

| 初值 | 入栈方式 | 栈顶指针指向 | 判满条件 |
|---|---|---|---|
| `top1=0, top2=MaxSize-1` | 先赋值再移动 | 栈顶元素的**下一个** | `top1 - 1 == top2` |
| `top1=-1, top2=MaxSize` | 先移动再赋值 | **栈顶元素本身** | `top1 + 1 == top2` |

> **做任何共享栈的题，先看初值和入栈方式，推出栈顶指针指向哪，再画一次「刚好满」的图。** 不要背条件。

#### 两个栈模拟队列（老师说出题概率比较大）

```c
int EnQ(Stk *S1, Stk *S2, ElemType x) {
    if (!StkFull(S1)) { StkPush(S1, x); return 1; }     /* ① S1 不满 → 直接入 */
    if (!StkEmpty(S2)) return 0;                        /* ② S1 满且 S2 非空 → 队满，拒绝 */
    ElemType t;                                          /* ③ S1 满且 S2 空 → 倒腾后再入 */
    while (!StkEmpty(S1)) { StkPop(S1, &t); StkPush(S2, t); }
    StkPush(S1, x);
    return 1;
}
int DeQ(Stk *S1, Stk *S2, ElemType *x) {
    if (!StkEmpty(S2)) return StkPop(S2, x);            /* ① S2 非空 → 直接出 */
    if (StkEmpty(S1)) return 0;                         /* ② 两个都空 → 队空 */
    ElemType t;                                          /* ③ S2 空、S1 非空 → 倒腾后出 */
    while (!StkEmpty(S1)) { StkPop(S1, &t); StkPush(S2, t); }
    return StkPop(S2, x);
}
int QEmpty(Stk *S1, Stk *S2) { return StkEmpty(S1) && StkEmpty(S2); }
```

**三条判据（必须说得出理由）**：
1. **队满** = S1 满 **且** S2 非空。因为 S2 非空时不能把 S1 倒进去，否则出队顺序全乱
2. **队空** = S1 和 S2 **都空**
3. S1 只管入，S2 只管出，**倒腾只在 S2 空的时候做**

#### 栈和队列能出的题，就这四道

① 共享栈 ② 两栈模拟队列 ③ 括号匹配 ④ 用栈逆置队列

【老师原话】「中缀转后缀、后缀求值**考代码题的可能性几乎没有**，因为它规则多、更适合出选择题。括号匹配反过来 —— 选择题不好出，代码又不难，所以它是有可能的。」

### 12.4 树（P19–P26）

**真题分布**：14、17、22 三年。【老师原话】「树虽然考得不多，但**考出来的题目其实都不是非常难的题目**，大家合理安排自己的时间。」

#### 四步解题流程（老师在 P26 的总结）

```
① 先确定用「层次遍历」还是「递归」
② 递归的话，确定用 先 / 中 / 后 哪一个更方便
③ 左右子树递归调用，确定要传什么参数
④ 想清楚「处理根」这一步要做什么   ← 主要部分全在这一步
```

【老师原话】「这四道题里：第一题的根做了 swap，第二题的根做了建结点+确定范围，第三题的根做了 count++ 和 if，第四题的根做了判断左右孩子是否为 x。**主要的部分都是根，但不要把这部分想复杂了 —— 只需要思考当前这个根需要做什么。**」

#### 层次 vs 递归的判据

| 题干出现 | 用什么 |
|---|---|
| 层次 / 高度 / 深度 / 宽度 / 完全二叉树 / 同一层 | **层次遍历** |
| 其他 | **优先递归** |
| —— | **非递归遍历是最后才考虑的** |

【老师原话】「先序中序后序是**从上到下再回过头**的路径，没办法横着看；层次是每一层访问完再下一层。所以**只要问题出现在『同一层』上，就一定用层次遍历。**」

#### 先序非递归的简单版（比"一路向左"好记得多）

```c
void PreNoRec(BiTree T) {
    if (!T) return;
    BiTNode *st[MaxSize]; int top = -1;
    st[++top] = T;
    while (top != -1) {
        BiTNode *p = st[top--];
        visit(p);
        if (p->rchild) st[++top] = p->rchild;    /* ★ 先压右 */
        if (p->lchild) st[++top] = p->lchild;    /* ★ 后压左（左才先出栈） */
    }
}
```

#### 后序非递归 = 先序改造 + 逆序（老师推荐的最好记法）

```c
void PostNoRec(BiTree T) {
    if (!T) return;
    BiTNode *st[MaxSize]; int top = -1;
    ElemType buf[MaxSize]; int n = 0;
    st[++top] = T;
    while (top != -1) {
        BiTNode *p = st[top--];
        buf[n++] = p->data;                      /* ★ 不 printf，存进数组 */
        if (p->lchild) st[++top] = p->lchild;    /* ★ 与先序相反：先压左 */
        if (p->rchild) st[++top] = p->rchild;
    }
    while (n > 0) visit(buf[--n]);               /* ★ 倒着输出 */
}
```
> **原理**：把「根左右」的压栈顺序反过来得到「根右左」，再整体逆序 = 「左右根」。
> 【老师原话】「后序遍历有好几种实现（两个栈、标志位）都比较复杂不好理解，而这个**只需要在先序基础上调换两行 + 加一个数组**。」

【老师原话】「**往年真题从来没有考过遍历的非递归实现。** 大家量力而行。能掌握递归就掌握递归，不要为了掌握非递归把递归也忘了。」

#### 层次遍历求最大宽度

```c
int MaxWidth(BiTree T) {
    if (!T) return 0;
    BiTNode *Q[MaxSize]; int front = 0, rear = 0, mx = 0;
    Q[rear++] = T;
    while (front < rear) {
        int size = rear - front;                 /* ★ 本层结点数，命门就是这一行 */
        if (size > mx) mx = size;
        for (int i = 0; i < size; i++) {         /* ★ 把本层全部出队，孩子全部入队 */
            BiTNode *p = Q[front++];
            if (p->lchild) Q[rear++] = p->lchild;
            if (p->rchild) Q[rear++] = p->rchild;
        }
    }
    return mx;
}
```
> 关键在于**「本层全部出队完毕」的那个瞬间，队列里恰好就是下一层的全部结点**。用 `size` 一次性出完这一层，就锁住了那个瞬间。

#### 判断完全二叉树（唯一需要「空结点也入队」的题）

```c
int IsComplete(BiTree T) {
    if (!T) return 1;
    BiTNode *Q[MaxSize]; int front = 0, rear = 0;
    int metNull = 0;
    Q[rear++] = T;
    while (front < rear) {
        BiTNode *p = Q[front++];
        if (p == NULL) { metNull = 1; continue; }   /* 记下遇到过空 */
        if (metNull) return 0;                       /* ★ 空之后又出现非空 → 不是完全二叉树 */
        Q[rear++] = p->lchild;                       /* ★ 空也入队（普通层序是判空才入） */
        Q[rear++] = p->rchild;
    }
    return 1;
}
```

#### 递归题的三个难度层级（区别在「特殊结点」有几种）

```c
/* ① 只有空结点是特殊的 */
int Count(BiTree T) {
    if (!T) return 0;
    return 1 + Count(T->lchild) + Count(T->rchild);
}

/* ② 空结点 + 叶结点 —— 叶结点 return 1（因为它没有孩子了） */
int CountLeaf(BiTree T) {
    if (!T) return 0;
    if (!T->lchild && !T->rchild) return 1;
    return CountLeaf(T->lchild) + CountLeaf(T->rchild);
}

/* ③ 空结点 + 双分支结点 —— 双分支 return 1 + 左 + 右（★ 它还有孩子！） */
int CountDouble(BiTree T) {
    if (!T) return 0;
    if (T->lchild && T->rchild)
        return 1 + CountDouble(T->lchild) + CountDouble(T->rchild);   /* ★ 不能只 return 1 */
    return CountDouble(T->lchild) + CountDouble(T->rchild);
}
```
【老师原话】「求叶结点时可以直接 `return 1`，因为叶结点没有孩子了；但**双分支结点是有左右子树的，必须 `return 1 + 左 + 右`。这两个题目是有区别的，不能无脑写 return 1。**」

**求高度 / 求最小值** 同理：
```c
int Height(BiTree T) {
    if (!T) return 0;
    int l = Height(T->lchild), r = Height(T->rchild);
    return 1 + (l > r ? l : r);          /* ★ 取 max 不是 sum，也不是 min */
}
```
> 【老师原话】关于 `max` 函数：**可以直接写 `max(a,b)`，老师能读懂**；不放心就用三目 `a>b?a:b`。三个数取最值要嵌套三目。

#### 交换左右子树 —— 中序万万不可

【老师原话】「后序最合适，先序也可以，**中序万万不可**。」

**理由**：中序是「左—根—右」。处理根时已经把左右子树交换了，接着去遍历「右子树」，实际走的是**原来的左子树**（它刚被换到右边），原来的右子树反而永远不会被处理。

```c
void SwapChild(BiTree T) {
    if (!T) return;
    SwapChild(T->lchild); SwapChild(T->rchild);      /* 后序：先处理完两个孩子 */
    BiTNode *t = T->lchild; T->lchild = T->rchild; T->rchild = t;
}
```

#### 递归参数什么时候要加 `*`（非常实用的判断标准）

```c
/* 求先序遍历第 k 个结点 —— count 必须是指针 */
void PreKth(BiTree T, int k, int *count, ElemType *res) {
    if (!T || *count >= k) return;
    (*count)++;
    if (*count == k) { *res = T->data; return; }
    PreKth(T->lchild, k, count, res);
    PreKth(T->rchild, k, count, res);
}
```

| 这个量的语义 | 怎么传 | 例子 |
|---|---|---|
| 需要随递归**持续累加、不能回退** | **`int *` 或全局变量** | 计数器 count、已找到的答案 |
| 需要**回到上一层的值** | **普通值传递** | 深度 deep、剩余长度 k |

> 【老师原话】「有很多参数是我们要根据当前内容保存下来、回来的时候还要用这个 —— 那就不要加星，直接用局部变量。但如果像这个 count 一样每次都要递增，就要写成星。」

#### 求值为 x 的结点的双亲

```c
void FindParent(BiTree T, ElemType x, BiTNode **p) {
    if (!T || *p) return;
    if ((T->lchild && T->lchild->data == x) || (T->rchild && T->rchild->data == x)) { *p = T; return; }
    FindParent(T->lchild, x, p);
    if (*p == NULL) FindParent(T->rchild, x, p);     /* ★ 左边找到了就不找右边 */
}
```

#### 表达式树求值（题型四）

```c
int EvalExprTree(BiTree T) {
    if (!T) return 0;
    if (!T->lchild && !T->rchild) return T->data - '0';   /* ★ 叶结点 = 操作数，char 转 int */
    int l = EvalExprTree(T->lchild), r = EvalExprTree(T->rchild);
    switch (T->data) {                                     /* ★ 非叶 = 运算符，必须 switch */
        case '+': return l + r;
        case '-': return l - r;
        case '*': return l * r;
        case '/': return r ? l / r : 0;
    }
    return 0;
}
```
> **data 必须是 `char`**（既要存数字又要存运算符），所以取数字要 `- '0'`。
> **表达式树的题一律用中序**，特殊处理放在「处理根」那一步。

### 12.5 图（P27–P32）

**真题分布**：**2009–2020 一次都没考过图**，2021（EL 路径）、2023（K 顶点）、2024（拓扑唯一性）连考三年。【老师原话】「2021 和 2023 其实都是**求度**。在 2024 拓扑序列出现之前，我们认为图能出的难度最多也就到度了。」

#### 明确的优先级（老师在 P32 给的）

| 优先级 | 内容 | 老师原话 |
|---|---|---|
| **保底分** | ① 存储结构 ② 求度 ③ DFS + BFS | 「一定要像背诗一样就能写出来」 |
| **★★** | 路径问题（5 个变体） | 「不是必考，但考的话也就考到这个难度」 |
| **★★** | 连通性、判环、判树 | 「非常经典和有用」 |
| 变态才到 | Prim / Kruskal / Dijkstra / Floyd | 「重要的是知道过程和思想」 |

#### 求度（考得最多，必须会四种组合）

|  | 无向图 | 有向图 |
|---|---|---|
| **邻接矩阵** | 第 i 行（或第 i 列）非零个数 | 出度 = 第 i **行**；入度 = 第 i **列** |
| **邻接表** | 第 i 个顶点的边表结点个数 | 出度 = 边表结点个数；**入度 = 扫描全部边表数 `adjvex == i`**（O(n+e)） |

> 0/1 邻接矩阵求出度可以**直接把整行加起来**，不用判断。带权图就要 `if (Edge[i][j] != 0) d++;`

#### 路径问题的五道题（老师说练这五道就够了）

| 题 | 用 BFS 还是 DFS | 要不要回溯 |
|---|---|---|
| 1. 输出 u→v 的简单路径 | BFS + parent | 否 |
| 2. u→v 的**最短**路径 | **必须 BFS**（层次性保证先遇到即最短） | 否 |
| 3. 从 v 出发**最远**的顶点 | BFS + dist 数组 | 否 |
| 4. 判断两点间**是否有**路径 | 都行 | 否 |
| 5. 长度为 **K** 的简单路径 / **所有**简单路径 | **必须 DFS** | **是** |

**核心手法：BFS + parent 数组 + 递归打印**
```c
/* parent 是反向的，用递归可以正序输出 */
void PrintPath(int par[], int s, int t, int out[], int *n) {
    if (par[t] != -1 && s != t) PrintPath(par, s, par[t], out, n);
    out[(*n)++] = t;
}

int BFS_Path(MGraph *G, int u, int v, int out[]) {
    if (u == v) { out[0] = u; return 1; }               /* 起点即终点，先特判 */
    for (int i = 0; i < G->numVertices; i++) { visited[i] = 0; parent[i] = -1; }
    int Q[MAXV], f = 0, r = 0;
    Q[r++] = u; visited[u] = 1;
    while (f < r) {
        int cur = Q[f++];
        for (int k = 0; k < G->numVertices; k++)
            if (G->Edge[cur][k] && !visited[k]) {
                visited[k] = 1;
                parent[k] = cur;                         /* ★ 先访问的是双亲，后访问的是孩子 */
                if (k == v) { int n = 0; PrintPath(parent, u, v, out, &n); return n; }
                Q[r++] = k;
            }
    }
    return 0;                                            /* 不可达 */
}
```
> `PrintPath` 那三行是整个路径专题的核心技巧：**先递归到起点，回溯时才输出** —— 反向的 parent 链就正着打出来了。

**求最远顶点**：把 parent 换成 dist
```c
dist[u] = dist[t] + 1;                   /* ★ 孩子的距离 = 双亲 + 1 */
if (dist[u] > dist[mx]) mx = u;          /* mx 始终记录最远的那个 */
```

#### 回溯：什么时候需要、需要撤销什么

【老师原话】「**所有涉及到『重新找』的都需要做回溯。**」

**判据**：这道题需不需要让**同一个顶点被访问多次**？
- 需要（长度为 K 的路径走错要重走；所有路径中一个顶点会出现在多条路径上）→ **要回溯**
- 不需要（最短路径、最远顶点、是否连通，每个顶点访问一次就够）→ **不回溯**

```c
/* 是否存在 i→j 长度为 k 的简单路径 */
int DFS_K(MGraph *G, int i, int j, int k) {
    if (i == j && k == 0) return 1;                      /* 到终点 且 长度正好 */
    if (k <= 0) return 0;
    visited[i] = 1;
    for (int u = 0; u < G->numVertices; u++)
        if (G->Edge[i][u] && !visited[u])
            if (DFS_K(G, u, j, k - 1)) { visited[i] = 0; return 1; }  /* ★ 子调用成功立刻上传 */
    visited[i] = 0;                                      /* ★ 回溯：撤销标记 */
    return 0;
}

/* 输出 u→v 的所有简单路径 */
int path[MAXV], d = 0;
void FindAllPath(MGraph *G, int u, int v) {
    visited[u] = 1;
    path[d++] = u;
    if (u == v) { for (int i = 0; i < d; i++) printf("%d ", path[i]); printf("\n"); }
    else
        for (int i = 0; i < G->numVertices; i++)
            if (G->Edge[u][i] && !visited[i]) FindAllPath(G, i, v);
    d--; visited[u] = 0;                                 /* ★ 回溯要撤销两件事 */
}
```
> **回溯的两件事必须都做**：`visited[u] = 0`（让这个顶点还能被别的路径用）和 `d--`（把 path 里的它弹掉）。
> 只撤 visited 不撤 d → 路径打印错乱；只撤 d 不撤 visited → 只能找到一条路径。
>
> **为什么不能边走边 printf？** 因为往前走的时候不知道这条路最终能不能到终点，printf 出去就撤不回来了。所以先存进 `path`，**到达终点才输出**。

#### 判连通 / 判环 / 判树

```c
/* 连通：只调用一次 DFS，看是不是全访问到了 */
int Connected(MGraph G) {
    for (int i = 0; i < G.numVertices; i++) visited[i] = 0;
    DFS(G, 0);
    for (int i = 0; i < G.numVertices; i++) if (!visited[i]) return 0;
    return 1;
}

/* 无向图判环：DFS 传父结点，遇到「已访问 且 不是父结点」的邻接点 */
int HasCycle(MGraph G, int v, int parent) {
    visited[v] = 1;
    for (int u = 0; u < G.numVertices; u++) {
        if (!G.Edge[v][u]) continue;              /* ★ 这里不能提前用 !visited[u] 过滤 */
        if (!visited[u]) { if (HasCycle(G, u, v)) return 1; }
        else if (u != parent) return 1;
    }
    return 0;
}
```
- **判树 = 无环 + 连通**，把两个函数合起来
- **有向图判环**：拓扑排序后 `count != 顶点数`（代码长，除非本来就考拓扑排序）

#### Kruskal 需要换存储结构（灵活变通的范例）

【老师原话】「无论用邻接矩阵还是邻接表，**要在整个边集合中找最小权值都很麻烦**。我们完全可以根据算法需求更改图的存储结构，**大家一定要懂得灵活变通**。」

```c
typedef struct { int weight; int begin, end; } Edge;     /* 边集数组 */
typedef struct {
    int vexnum, arcnum;
    Edge adjs[MAXE];              /* 按 weight 排序后逐条取 */
    char vertex[MAXV];
} EdgeGraph;
```
- **Kruskal 的核心是并查集**。老师说「并查集虽然没考过，但在 Kruskal 中是非常重要的应用」
- 排序那一步可以直接写 `sort(G.adjs, ...)` 并注明「按 weight 升序」，**不必实现排序**

#### 两条容易漏的

1. **拓扑排序的辅助栈里，元素始终都是入度为 0 的点** —— 栈或队列都行，用栈得到的序列和用队列不同，都是合法拓扑序列
2. **DFS 也能生成拓扑序列**：把 visit 从「进入时输出」改成「**DFS 结束后存入 result 数组**」→ 得到**逆拓扑序列**，倒过来就是拓扑序列

### 12.6 查找与排序（P33–P34）

#### 建堆的起始位置（高频错点）

| 数组起始下标 | 孩子公式 | 建堆从哪开始 |
|---|---|---|
| 从 **0** 开始 | `2i+1` / `2i+2` | `i = (n-2)/2` |
| 从 **1** 开始（教材） | `2i` / `2i+1` | `i = n/2` |

```c
void SiftDown0(int A[], int k, int n) {              /* A[0..n-1] 大根堆 */
    int tmp = A[k];
    for (int i = 2*k + 1; i < n; i = 2*i + 1) {
        if (i + 1 < n && A[i] < A[i+1]) i++;         /* 取较大的孩子 */
        if (tmp >= A[i]) break;
        A[k] = A[i]; k = i;
    }
    A[k] = tmp;
}
void BuildHeap0(int A[], int n) { for (int i = (n-2)/2; i >= 0; i--) SiftDown0(A, i, n); }
```

#### Top-K：堆的方向和直觉相反

| 要找 | 建什么堆 | 大小 |
|---|---|---|
| 前 k 个**最大** | **小根堆** | k |
| 前 k 个**最小** | **大根堆** | k |

**为什么**：找前 k 大时，堆里装的是「目前最大的 k 个」。要淘汰的是**这 k 个里最小的那个** —— 所以堆顶必须是最小的，即小根堆。

```c
void TopKMax(int A[], int n, int k, int heap[]) {
    for (int i = 0; i < k; i++) heap[i] = A[i];
    for (int i = (k-2)/2; i >= 0; i--) SiftDownMin(heap, i, k);   /* 建小根堆 */
    for (int i = k; i < n; i++)
        if (A[i] > heap[0]) {                         /* 比堆顶（当前最小）大 → 顶掉它 */
            heap[0] = A[i];
            SiftDownMin(heap, 0, k);
        }
}
```

#### 简单选择排序的真正价值

【老师原话】「简单选择是**代码题中用得最多的**。因为大多数情况我们是要找最大或最小值，**只走它的内层循环就行了** —— O(n) 就能找到最值。中间那一部分是可以重复使用的。」

#### 实在没思路时

【老师原话】「最基本的排序算法一定要掌握，这样如果是顺序表和链表的题目，**大家可以直接怼上一个排序算法，可能也会得那么一点点的分数**。这是迫不得已的情况下。」

### 12.7 串（P35–P36）

【老师原话】「**KMP 不会考代码题。** 如果代码题考了字符串匹配，我们也可以考虑用暴力解法，**免得在前期浪费很多时间去理解 KMP 这个很难理解的算法**。」

#### 串的题型 = 顺序表的题型

【老师原话】「字符串按字符数组来讲，**和我们数字的顺序表的一维数组是一样的**。逆置、重排列，所有顺序表可能考察的题型都有可能换成串。这种题目直接把顺序表那部分的代码模板用到串上就行，没有什么区别。」

#### 暴力串匹配（三指针）

```c
int BF_Index(char S[], char T[]) {
    int slen = strlen(S), tlen = strlen(T);
    for (int i = 0; i + tlen <= slen; i++) {     /* i = 本轮的起始位置，不参与移动 */
        int k = i, j = 0;                        /* ★ k 是临时指针，替 i 往后走 */
        while (j < tlen && S[k] == T[j]) { k++; j++; }
        if (j == tlen) return i + 1;             /* 子串走完 = 匹配成功 */
    }
    return 0;
}
```
> **为什么要第三个指针 k**：主串指针 `i` 是「本轮从哪开始比」，失配后 `i` 只前进一位。如果直接用 `i` 往后走，就丢失了起始位置。**所有「起点 + 往后比」型的题都是这个三指针结构。**

#### 最长重复子串（同一个三指针结构）

```c
int LongestRepeat(char s[], int *start) {
    int len = strlen(s), maxlen = 0;
    *start = -1;
    for (int i = 1; i < len; i++)
        for (int j = 0; j < i; j++) {            /* ★ j 只到 i，否则 i==j 恒相等 */
            int k = 0;
            while (i + k < len && s[j+k] == s[i+k]) k++;
            if (k > maxlen) { maxlen = k; *start = j; }
        }
    return maxlen;
}
```

#### C 库里现成的
`#include <string.h>` 后可以用 `strstr(haystack, needle)` —— 返回首次出现位置的指针，没找到返回 NULL。**实际工作中用这个；考研是选拔性考试才要你手写。**
